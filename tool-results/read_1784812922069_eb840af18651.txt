   307→   307→function ChatHistoryList({
   308→   308→  messages,
   309→   309→  onLoadMore,
   310→   310→  hasMore,
   311→   311→}: {
   312→   312→  messages: ChatMsg[];
   313→   313→  onLoadMore: () => void;
   314→   314→  hasMore: boolean;
   315→   315→}) {
   316→   316→  // Group messages by date
   317→   317→  const grouped = messages.reduce<Record<string, ChatMsg[]>>((acc, msg) => {
   318→   318→    const date = new Date(msg.timestamp).toLocaleDateString("ru-RU", {
   319→   319→      day: "numeric",
   320→   320→      month: "long",
   321→   321→    });
   322→   322→    if (!acc[date]) acc[date] = [];
   323→   323→    acc[date].push(msg);
   324→   324→    return acc;
   325→   325→  }, {});
   326→   326→
   327→   327→  return (
   328→   328→    <div className="space-y-4">
   329→   329→      {Object.entries(grouped).map(([date, msgs]) => (
   330→   330→        <div key={date}>
   331→   331→          <div className="text-[10px] text-muted-foreground uppercase tracking-wider mb-2">
   332→   332→            {date}
   333→   333→          </div>
   334→   334→          <div className="space-y-2">
   335→   335→            {msgs.map((msg) => (
   336→   336→              <div
   337→   337→                key={msg.id}
   338→   338→                className="flex items-start gap-2 p-2 rounded-md hover:bg-muted/50 cursor-pointer"
   339→   339→              >
   340→   340→                <div className="shrink-0 mt-0.5">
   341→   341→                  {msg.role === "user" ? (
   342→   342→                    <MessageSquare className="h-3 w-3 text-muted-foreground" />
   343→   343→                  ) : (
   344→   344→                    <Bot className="h-3 w-3 text-primary" />
   345→   345→                  )}
   346→   346→                </div>
   347→   347→                <div className="flex-1 min-w-0">
   348→   348→                  <p className="text-xs line-clamp-2">{msg.content}</p>
   349→   349→                  <span className="text-[10px] text-muted-foreground">
   350→   350→                    {new Date(msg.timestamp).toLocaleTimeString("ru-RU", {
   351→   351→                      hour: "2-digit",
   352→   352→                      minute: "2-digit",
   353→   353→                    })}
   354→   354→                  </span>
   355→   355→                </div>
   356→   356→              </div>
   357→   357→            ))}
   358→   358→          </div>
   359→   359→        </div>
   360→   360→      ))}
   361→   361→      {hasMore && (
   362→   362→        <Button variant="ghost" size="sm" onClick={onLoadMore} className="w-full text-xs">
   363→   363→          Загрузить ещё
   364→   364→        </Button>
   365→   365→      )}
   366→   366→    </div>
   367→   367→  );
   368→   368→}

   428→   428→        total: number;

   464→   464→      timestamp: Date.now(),
   465→   465→    };

   467→   467→
   468→   468→    // Create streaming assistant message
   469→   469→    const streamMsgId = (Date.now() + 1).toString();
   470→   470→    const streamMsg: ChatMsg = {
   471→   471→      id: streamMsgId,
   472→   472→      role: "assistant",
   473→   473→      content: "",
   474→   474→      timestamp: Date.now(),
   475→   475→      isStreaming: true,
   476→   476→    };
   477→   477→    setMessages((prev) => [...prev, streamMsg]);
   478→   478→
   479→   479→    try {
   480→   480→      const baseUrl = getApiBaseUrl();
   481→   481→      const abortController = new AbortController();
   482→   482→      abortControllerRef.current = abortController;
   483→   483→
   484→   484→      const response = await fetch(`${baseUrl}/assistant/chat/stream`, {
   485→   485→        method: "POST",
   486→   486→        headers: {
   487→   487→          "Content-Type": "application/json",
   488→   488→          ...(token ? { Authorization: `Bearer ${token}` } : {}),
   489→   489→        },
   490→   490→        body: JSON.stringify({
   491→   491→          message: msg,
   492→   492→          project_id: projectId || undefined,
   493→   493→          context: undefined,
   494→   494→        }),
   495→   495→        signal: abortController.signal,
   496→   496→      });
   497→   497→
   498→   498→      if (!response.ok) {
   499→   499→        throw new Error(`HTTP ${response.status}: ${response.statusText}`);

   528→   528→                  prev.map((m) =>
   529→   529→                    m.id === streamMsgId
   530→   530→                      ? { ...m, content: accumulatedContent, isStreaming: true }
   531→   531→                      : m
   532→   532→                  )
   533→   533→                );
   534→   534→              } else if (event.type === "done") {

   537→   537→                  prev.map((m) =>
   538→   538→                    m.id === streamMsgId
   539→   539→                      ? {
   540→   540→                          ...m,
   541→   541→                          content: accumulatedContent,
   542→   542→                          isStreaming: false,
   543→   543→                          metadata: {
   544→   544→                            model_used: event.model_used || "",
   545→   545→                            provider: event.provider || "",
   546→   546→                            latency_ms: event.latency_ms || 0,
   547→   547→                          },
   548→   548→                        }
   549→   549→                      : m
   550→   550→                  )
   551→   551→                );
   552→   552→              }
   553→   553→            } catch {

   555→   555→              accumulatedContent += jsonStr;
   556→   556→              setMessages((prev) =>
   557→   557→                prev.map((m) =>
   558→   558→                  m.id === streamMsgId
   559→   559→                    ? { ...m, content: accumulatedContent, isStreaming: true }
   560→   560→                    : m
   561→   561→                )
   562→   562→              );
   563→   563→            }
   564→   564→          }
   565→   565→        }
   566→   566→      }
   567→   567→
   568→   568→      // Ensure message is finalized
   569→   569→      setMessages((prev) =>
   570→   570→        prev.map((m) =>
   571→   571→          m.id === streamMsgId ? { ...m, isStreaming: false } : m
   572→   572→        )
   573→   573→      );
   574→   574→    } catch (err) {

   578→   578→          prev.map((m) =>

   581→   581→        );

   585→   585→          const data = await apiFetch<{
   586→   586→            reply: string;
   587→   587→            model_used: string;
   588→   588→            provider: string;
   589→   589→            latency_ms: number;
   590→   590→          }>("/assistant/chat", {

   592→   592→            headers: { "Content-Type": "application/json" },

   598→   598→          });
   599→   599→
   600→   600→          setMessages((prev) =>
   601→   601→            prev.map((m) =>
   602→   602→              m.id === streamMsgId
   603→   603→                ? {
   604→   604→                    ...m,
   605→   605→                    content: data.reply,
   606→   606→                    isStreaming: false,
   607→   607→                    metadata: {
   608→   608→                      model_used: data.model_used,
   609→   609→                      provider: data.provider,
   610→   610→                      latency_ms: data.latency_ms,
   611→   611→                    },

   619→   619→              m.id === streamMsgId

   633→   633→      setIsStreaming(false);
   634→   634→      abortControllerRef.current = null;
   635→   635→    }
