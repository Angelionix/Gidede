     1→/**
     2→ * Gidede — In-memory store for AI Assistant (Block 7).
     3→ *
     4→ * The Prisma schema has no ChatMessage / AssistantSuggestion / AssistantAlert
     5→ * tables. Another agent is actively editing the schema, so we DON'T modify it.
     6→ * Instead we keep chat history + ephemeral state in module-level Maps keyed by
     7→ * `userId` (or `userId:projectId` when project_id is provided).
     8→ *
     9→ * Trade-off: data is reset on server restart — acceptable for a demo / preview.
    10→ */
    11→
    12→export interface ChatMsg {
    13→  id: string;
    14→  role: "user" | "assistant" | "system";
    15→  content: string;
    16→  timestamp: number;
    17→  metadata?: Record<string, unknown>;
    18→  project_id?: string | null;
    19→}
    20→
    21→/** Key: `${userId}` or `${userId}:${projectId}` */
    22→const chatHistory = new Map<string, ChatMsg[]>();
    23→
    24→function keyFor(userId: string, projectId?: string | null): string {
    25→  return projectId ? `${userId}:${projectId}` : userId;
    26→}
    27→
    28→export function getHistory(
    29→  userId: string,
    30→  projectId: string | null | undefined,
    31→  limit = 50
    32→): { messages: ChatMsg[]; total: number } {
    33→  const k = keyFor(userId, projectId || undefined);
    34→  const all = chatHistory.get(k) || [];
    35→  // Most-recent first → reverse chronological; we slice the latest `limit`.
    36→  const sliced = all.slice(-limit).reverse();
    37→  return { messages: sliced, total: all.length };
    38→}
    39→
    40→export function appendMessage(
    41→  userId: string,
    42→  projectId: string | null | undefined,
    43→  msg: ChatMsg
    44→): void {
    45→  const k = keyFor(userId, projectId || undefined);
    46→  const list = chatHistory.get(k) || [];
    47→  list.push(msg);
    48→  // Cap to 1000 messages per key to avoid unbounded growth.
    49→  if (list.length > 1000) list.splice(0, list.length - 1000);
    50→  chatHistory.set(k, list);
    51→}
    52→
    53→export function clearHistory(
    54→  userId: string,
    55→  projectId: string | null | undefined
    56→): number {
    57→  const k = keyFor(userId, projectId || undefined);
    58→  const list = chatHistory.get(k) || [];
    59→  const n = list.length;
    60→  chatHistory.delete(k);
    61→  return n;
    62→}

   601→}

   603→/** Derive alerts from project's pipeline state. Empty stages → warnings. */
   604→export function deriveAlerts(snap: ProjectPipelineSnapshot): Alert[] {
   605→  const now = Date.now();
   606→  const alerts: Alert[] = [];
   607→
   608→  if (!snap.hasConcept) {
   609→    alerts.push({
   610→      id: `alert-concept-${now}`,
   611→      alert_type: "missing_stage",

   617→      timestamp: now,
   618→    });
   619→  }
   620→
   621→  if (snap.hasConcept && !snap.hasCoreLoop) {
   622→    alerts.push({
   623→      id: `alert-coreloop-${now}`,
   624→      alert_type: "missing_stage",
   625→      severity: "warning",
   626→      block_id: 2,

   630→      timestamp: now,
   631→    });
   632→  }
   633→
   634→  if (snap.hasCoreLoop && !snap.hasMda) {
   635→    alerts.push({
   636→      id: `alert-mda-${now}`,
   637→      alert_type: "missing_stage",
   638→      severity: "info",
   639→      block_id: 3,
   640→      title: "MDA-анализ не выполнен",
   641→      description: "Core Loop есть, но MDA-профиль не построен.",

   643→      timestamp: now,
   644→    });
   645→  }
   646→
   647→  if (snap.hasMda && !snap.hasBalance) {
   648→    alerts.push({
   649→      id: `alert-balance-${now}`,
   650→      alert_type: "missing_stage",
   651→      severity: "info",
   652→      block_id: 4,
   653→      title: "Балансировка не запущена",

   656→      timestamp: now,
   657→    });
   658→  }
   659→
   660→  if (snap.hasBalance && !snap.hasProgression) {
   661→    alerts.push({
   662→      id: `alert-progression-${now}`,
   663→      alert_type: "missing_stage",
   664→      severity: "info",
   665→      block_id: 5,
   666→      title: "Прогрессия не спроектирована",
   667→      description: "Баланс есть, но прогрессия не построена.",
   668→      suggestion: "Запустите алгоритм 3.5 — построит tier-модель и кривые.",
   669→      timestamp: now,
   670→    });
   671→  }
   672→
   673→  if (snap.hasProgression && !snap.hasEconomy) {
   674→    alerts.push({
   675→      id: `alert-economy-${now}`,
   676→      alert_type: "missing_stage",
   677→      severity: "info",
   678→      block_id: 5,

   682→      timestamp: now,
   683→    });
   684→  }
   685→
   686→  if (snap.hasEconomy && !snap.hasGdd) {
   687→    alerts.push({
   688→      id: `alert-gdd-${now}`,

   690→      severity: "warning",
