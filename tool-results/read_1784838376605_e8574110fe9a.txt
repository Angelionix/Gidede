    95→function classifyStructuralType(
    96→  mechanics: string[],
    97→  genre: string,
    98→  desiredLoopType: string | undefined,
    99→  steps: CoreStep[]
   100→) {
   101→  // Determine loop type
   102→  let type: string;
   103→  if (desiredLoopType && VALID_LOOP_TYPES.includes(desiredLoopType)) {
   104→    type = desiredLoopType;
   105→  } else {
   106→    type = GENRE_DEFAULT_LOOP_TYPE[genre] || "hybrid";
   107→  }
   108→
   109→  // Sub-type
   110→  let subType = "hybrid_engine";
   111→  if (type === "engine") {
   112→    // Pure engine if no consumables, braked if has drains
   113→    const hasConsumed = steps.some((s) => s.resources_consumed.length > 0);
   114→    subType = hasConsumed ? "braked_engine" : "pure_engine";
   115→  } else if (type === "economy") {
   116→    // Multi-currency if many resources, single otherwise
   117→    const currencies = new Set<string>();
   118→    for (const s of steps) {
   119→      s.resources_consumed.forEach((r) => currencies.add(r));
   120→      s.resources_produced.forEach((r) => currencies.add(r));
   121→    }
   122→    subType =
   123→      currencies.size >= 3 ? "multi_currency_economy" : "single_currency_economy";
   124→  } else if (type === "ecology") {
   125→    subType = "balanced_ecology";
   126→  } else {
   127→    // hybrid
   128→    subType = mechanics.length % 2 === 0 ? "hybrid_engine" : "hybrid_economy";
   129→  }
   130→
   131→  const hasBraking = type !== "engine" || subType === "braked_engine";
   132→
   133→  // Currencies (extracted from steps)
   134→  const currenciesSet = new Set<string>();
   135→  for (const s of steps) {
   136→    s.resources_consumed.forEach((r) => currenciesSet.add(r));
   137→    s.resources_produced.forEach((r) => currenciesSet.add(r));
   138→  }
   139→  const currencies = Array.from(currenciesSet);
   140→
   141→  // Resources (with class)
   142→  const resources = currencies.map((name) => ({
   143→    name,
   144→    class_:
   145→      type === "engine"
   146→        ? "core"
   147→        : type === "economy"
   148→          ? "currency"
   149→          : "balance_state",
   150→  }));
   151→
   152→  // Loops description
   153→  const loops = [
   154→    {
   155→      type: "inner",
   156→      description: `Micro loop: ${steps[0]?.action || "act"} → ${steps[1]?.action || "respond"} (${Math.round(
   157→        (steps[0]?.duration_estimate || 5) + (steps[1]?.duration_estimate || 5)
   158→      )}s)`,
   159→    },
   160→    {
   161→      type: "outer",
   162→      description: `Outer loop ties ${steps.length} steps into a ${type} structure`,
   163→    },
   164→  ];
   165→
   166→  // Risk assessment
   167→  const riskLevel =
   168→    type === "ecology"
   169→      ? "high"
   170→      : type === "hybrid"
   171→        ? "medium"
   172→        : "low";
   173→  const likelyPathologies: string[] = [];
   174→  if (type === "engine") likelyPathologies.push("runaway");
   175→  if (type === "ecology") likelyPathologies.push("stall", "oscillation");
   176→  if (type === "hybrid") likelyPathologies.push("brittleness");
   177→  if (steps.length > 7) likelyPathologies.push("triviality", "stagnation");
   178→  if (steps.length < 3) likelyPathologies.push("deadlock");
   179→
   180→  const mitigationSuggestions: string[] = [];
   181→  if (likelyPathologies.includes("runaway"))
   182→    mitigationSuggestions.push("Add a balancing sink to drain excess resources");
   183→  if (likelyPathologies.includes("stall"))
   184→    mitigationSuggestions.push("Ensure every pool has a faucet AND a drain");
   185→  if (likelyPathologies.includes("brittleness"))
   186→    mitigationSuggestions.push("Add redundant paths so failure of one mechanic doesn't break the loop");
   187→  if (likelyPathologies.includes("triviality"))
   188→    mitigationSuggestions.push("Reduce step count to 3-7 core verbs");
   189→  if (likelyPathologies.includes("deadlock"))
   190→    mitigationSuggestions.push("Add at least 3 distinct steps to break circular deadlocks");
   191→
   192→  return {
   193→    type,
   194→    sub_type: subType,
   195→    has_braking: hasBraking,
   196→    currencies,
   197→    resources,
   198→    loops,
   199→    risk_assessment: {
   200→      risk_level: riskLevel,
   201→      likely_pathologies: likelyPathologies,
   202→      mitigation_suggestions: mitigationSuggestions,
   203→    },
   204→  };
   205→}

   207→function buildSteps(
   208→  mechanics: string[],
   209→  customSteps: string[] | undefined,
   210→  type: string
   211→): CoreStep[] {
   212→  // If custom steps provided, use them
   213→  if (customSteps && customSteps.length > 0) {
   214→    return customSteps.slice(0, 10).map((action, i) => {
   215→      const mech = mechanics[i % Math.max(1, mechanics.length)] || "core_action";
   216→      const feedbackType: CoreStep["feedback_type"] =
   217→        i % 3 === 0 ? "positive" : i % 3 === 1 ? "negative" : "neutral";
   218→      return {
   219→        action,
   220→        mechanics: [mech],
   221→        resources_consumed: feedbackType === "negative" ? ["energy"] : [],
   222→        resources_produced: feedbackType === "positive" ? ["xp", "score"] : [],
   223→        feedback_type: feedbackType,
   224→        duration_estimate: 5 + (i % 5) * 2,
   225→      };
   226→    });
   227→  }
   228→
   229→  // Default 5-step loop
   230→  const m0 = mechanics[0] || "explore";
   231→  const m1 = mechanics[1] || "combat";
   232→  const m2 = mechanics[2] || "reward";
   233→  const m3 = mechanics[3] || "progress";
   234→  const m4 = mechanics[4] || "return";
   235→
   236→  return [
   237→    {
   238→      action: `Find target (${m0})`,
   239→      mechanics: [m0],
   240→      resources_consumed: [],
   241→      resources_produced: ["signal"],
   242→      feedback_type: "neutral",
   243→      duration_estimate: 6,
   244→    },
   245→    {
   246→      action: `Engage (${m1})`,
   247→      mechanics: [m1],
   248→      resources_consumed: ["energy", "ammo"],
   249→      resources_produced: [],
   250→      feedback_type: "negative",
   251→      duration_estimate: 10,
   252→    },
   253→    {
   254→      action: `Collect rewards (${m2})`,
   255→      mechanics: [m2],
   256→      resources_consumed: [],
   257→      resources_produced: ["xp", "gold"],
   258→      feedback_type: "positive",
   259→      duration_estimate: 4,
   260→    },
   261→    {
   262→      action: `Upgrade (${m3})`,
   263→      mechanics: [m3],
   264→      resources_consumed: ["gold"],
   265→      resources_produced: ["power", "ability"],
   266→      feedback_type: "positive",
   267→      duration_estimate: 8,
   268→    },
   269→    {
   270→      action: `Return to base (${m4})`,
   271→      mechanics: [m4],
   272→      resources_consumed: [],
   273→      resources_produced: ["rest", "save"],
   274→      feedback_type: "neutral",
   275→      duration_estimate: 5,
   276→    },
   277→  ];
   278→}

   290→        actions: steps.map((s) => s.action),
   291→        parent_step: "core_loop",

   295→      {
   296→        actions: ["Complete 3 core loops", "Trigger side activity"],
   297→        parent_step: "small_loop",
   298→      },
   299→    ],
   300→    large: [
   301→      {
   302→        actions: ["Complete quest arc", "Unlock new area"],
   303→        parent_step: "medium_loop",
   304→      },
   305→    ],
   306→    macro: [
   307→      {
   308→        actions: ["Reach level cap", "Beat final boss"],
   309→        parent_step: "large_loop",
   310→      },
   311→    ],
   312→    meta: [
   313→      {

   315→        parent_step: "macro_loop",
   316→      },
   317→    ],
   318→  };

   324→      parent_step: steps[2]?.action || "mid_step",

   326→  }

   331→function detectPathologies(
   332→  steps: CoreStep[],
   333→  structuralType: { type: string; risk_assessment: { likely_pathologies: string[] } }
   334→) {
   335→  const pathologies: Array<{
   336→    name: string;
   337→    type: string;
   338→    severity: string;
   339→    description: string;
   340→    correction: string;
   341→    affected_resources: string[];
   342→  }> = [];
   343→
   344→  const likely = structuralType.risk_assessment.likely_pathologies;
   345→
   346→  // Runaway: positive feedback dominates
   347→  const positiveCount = steps.filter((s) => s.feedback_type === "positive").length;
   348→  if (likely.includes("runaway") || positiveCount > steps.length / 2) {
   349→    pathologies.push({
   350→      name: "Runaway",
   351→      type: "runaway",
   352→      severity: positiveCount > steps.length * 0.6 ? "critical" : "warning",
   353→      description: `Loop has ${positiveCount}/${steps.length} positive-feedback steps — accumulation may run away`,
   354→      correction: "Add a balancing sink step that drains resources",
   355→      affected_resources: steps
   356→        .flatMap((s) => s.resources_produced)
   357→        .slice(0, 3),
   358→    });
   359→  }
   360→
   361→  // Deadlock: too few steps
   362→  if (likely.includes("deadlock") || steps.length < 3) {
   363→    pathologies.push({
   364→      name: "Deadlock",
   365→      type: "deadlock",
   366→      severity: "critical",
   367→      description: `Only ${steps.length} step(s) — risk of circular deadlock with no escape`,
   368→      correction: "Add at least 3 distinct steps",
   369→      affected_resources: [],
   370→    });
   371→  }
   372→
   373→  // Stall: no positive feedback
   374→  if (likely.includes("stall") || positiveCount === 0) {
   375→    pathologies.push({
   376→      name: "Stall",
   377→      type: "stall",
   378→      severity: "warning",
   379→      description: "Loop has no positive-feedback steps — players may stall",
   380→      correction: "Add a reward step with positive feedback",
   381→      affected_resources: [],
   382→    });
   383→  }
   384→
   385→  // Brittleness: only one path through the loop
   386→  if (likely.includes("brittleness") || steps.every((s) => s.mechanics.length <= 1)) {
   387→    pathologies.push({
   388→      name: "Brittleness",
   389→      type: "brittleness",
   390→      severity: "warning",
   391→      description: "Each step uses a single mechanic — single point of failure",
   392→      correction: "Add alternative mechanics to at least 2 steps",
   393→      affected_resources: [],
   394→    });
   395→  }
   396→
   397→  // Oscillation: alternating positive/negative
   398→  const feedbackPattern = steps.map((s) => s.feedback_type).join("-");
   399→  if (likely.includes("oscillation") || feedbackPattern.includes("positive-negative-positive")) {
   400→    pathologies.push({
   401→      name: "Oscillation",
   402→      type: "oscillation",
   403→      severity: "info",
   404→      description: "Feedback pattern alternates rapidly — may cause emotional whiplash",
   405→      correction: "Smooth the feedback curve by inserting neutral steps",
   406→      affected_resources: [],
   407→    });
   408→  }
   409→
   410→  // Stagnation: no progression steps
   411→  const hasProgression = steps.some((s) =>
   412→    s.mechanics.some((m) => m.includes("progress") || m.includes("upgrade") || m.includes("level"))
   413→  );
   414→  if (likely.includes("stagnation") || !hasProgression) {
   415→    pathologies.push({
   416→      name: "Stagnation",
   417→      type: "stagnation",
   418→      severity: "warning",
   419→      description: "No progression mechanic detected — long-term stagnation risk",
   420→      correction: "Add a progression step (level-up, unlock, upgrade)",
   421→      affected_resources: [],
   422→    });
   423→  }
   424→
   425→  // Triviality: too many steps
   426→  if (likely.includes("triviality") || steps.length > 7) {
   427→    pathologies.push({
   428→      name: "Triviality",
   429→      type: "triviality",
   430→      severity: "info",
   431→      description: `${steps.length} steps is above the 3-7 ideal — risk of diluting the core verb`,
   432→      correction: "Consolidate similar steps; aim for 3-7 core steps",
   433→      affected_resources: [],
   434→    });
   435→  }
   436→
   437→  // === Type-specific pathologies for new core loop types ===
   438→
   439→  const loopType = structuralType.type;
   440→
   441→  // Tower Defense specific pathologies
   442→  if (loopType === "tower_defense") {
   443→    // Wave imbalance: too few defense steps vs build steps
   444→    const buildSteps = steps.filter((s) =>
   445→      s.mechanics.some((m) => m.includes("build") || m.includes("place") || m.includes("upgrade"))
   446→    ).length;
   447→    const defendSteps = steps.filter((s) =>
   448→      s.mechanics.some((m) => m.includes("defend") || m.includes("shoot") || m.includes("attack"))
   449→    ).length;
   450→    if (defendSteps < buildSteps) {
   451→      pathologies.push({
   452→        name: "Wave Imbalance",
   453→        type: "wave_imbalance",
   454→        severity: "warning",
   455→        description: `Defense has ${defendSteps} defend vs ${buildSteps} build steps — players may over-invest in economy and ignore defense`,
   456→        correction: "Balance build:defend ratio closer to 1:1, or add urgency mechanics (timed waves)",
   457→        affected_resources: [],
   458→      });
   459→    }
   460→    // Snowball: no recovery mechanic
   461→    const hasRecovery = steps.some((s) =>
   462→      s.mechanics.some((m) => m.includes("repair") || m.includes("heal") || m.includes("recover"))
   463→    );
   464→    if (!hasRecovery) {
   465→      pathologies.push({
   466→        name: "No Recovery",
   467→        type: "no_recovery",
   468→        severity: "info",
   469→        description: "No repair/recovery mechanic — once base is damaged, players cannot bounce back",
   470→        correction: "Add a repair or shield regeneration step between waves",
   471→        affected_resources: [],
   472→      });
   473→    }
   474→  }
   475→
   476→  // Rhythm specific pathologies
   477→  if (loopType === "rhythm") {
   478→    // Off-beat penalty: too many negative feedback steps
   479→    const negativeCount = steps.filter((s) => s.feedback_type === "negative").length;
   480→    if (negativeCount > steps.length / 2) {
   481→      pathologies.push({
   482→        name: "Off-Beat Penalty",
   483→        type: "off_beat_penalty",
   484→        severity: "warning",
   485→        description: `${negativeCount}/${steps.length} steps have negative feedback — rhythm feels punishing, not rewarding`,
   486→        correction: "Increase positive feedback for successful hits; reduce miss penalty severity",
   487→        affected_resources: [],
   488→      });
   489→    }
   490→    // Tempo drift: no calibration step
   491→    const hasCalibration = steps.some((s) =>
   492→      s.mechanics.some((m) => m.includes("calibrate") || m.includes("sync") || m.includes("tempo"))
   493→    );
   494→    if (!hasCalibration && steps.length > 4) {
   495→      pathologies.push({
   496→        name: "Tempo Drift",
   497→        type: "tempo_drift",
   498→        severity: "info",
   499→        description: "No tempo calibration step — difficulty may drift unevenly across the song",
   500→        correction: "Add a tempo-sync or BPM-shift step to maintain consistent challenge curve",
   501→        affected_resources: [],
   502→      });
   503→    }
   504→  }
   505→
   506→  // Puzzle specific pathologies
   507→  if (loopType === "puzzle") {
   508→    // Stuck state: no hint or reset mechanic
   509→    const hasHint = steps.some((s) =>
   510→      s.mechanics.some((m) => m.includes("hint") || m.includes("reset") || m.includes("undo") || m.includes("clear"))
   511→    );
   512→    if (!hasHint) {
   513→      pathologies.push({
   514→        name: "Stuck State",
   515→        type: "stuck_state",
   516→        severity: "critical",
   517→        description: "No hint/undo/reset mechanic — players who make a mistake cannot recover, leading to frustration",
   518→        correction: "Add an undo step, hint system, or board-reset mechanic",
   519→        affected_resources: [],
   520→      });
   521→    }
   522→    // Pattern blindness: too many piece types
   523→    const pieceTypes = steps.filter((s) =>
   524→      s.mechanics.some((m) => m.includes("piece") || m.includes("shape") || m.includes("block"))
   525→    ).length;
   526→    if (pieceTypes > 4) {
   527→      pathologies.push({
   528→        name: "Pattern Blindness",
   529→        type: "pattern_blindness",
   530→        severity: "warning",
   531→        description: `${pieceTypes} piece-related steps — cognitive overload, players can't recognize patterns`,
   532→        correction: "Limit piece variety to 3-4 types; introduce complexity gradually",
   533→        affected_resources: [],
   534→      });
   535→    }
   536→  }
   537→
   538→  return {
   539→    pathologies,
   540→    total_count: pathologies.length,
   541→    critical_count: pathologies.filter((p) => p.severity === "critical").length,
   542→  };
   543→}

   545→function buildValidation(
   546→  steps: CoreStep[],
   547→  pathologies: { pathologies: Array<{ severity: string }>; critical_count: number },
   548→  structuralType: { type: string; has_braking: boolean }
   549→) {
   550→  // Fun check: passed if positive feedback exists and step count is reasonable
   551→  const positiveCount = steps.filter((s) => s.feedback_type === "positive").length;
   552→  const funCheckScore = Math.min(
   553→    1,
   554→    (positiveCount / Math.max(1, steps.length)) * 0.5 +
   555→      (steps.length >= 3 && steps.length <= 7 ? 0.5 : 0.2)
   556→  );
   557→  const funCheckPassed = funCheckScore >= 0.5;
   558→
   559→  // Loop closedness: always true since we close the loop
   560→  const loopClosedness = {
   561→    is_closed: true,
   562→    connection_description: `Last step "${steps[steps.length - 1]?.action || "return"}" feeds back into first step "${steps[0]?.action || "find"}"`,
   563→  };
   564→
   565→  // Resource sufficiency
   566→  const allConsumed = new Set(steps.flatMap((s) => s.resources_consumed));
   567→  const allProduced = new Set(steps.flatMap((s) => s.resources_produced));
   568→  const deadResources = Array.from(allProduced).filter(
   569→    (r) => !allConsumed.has(r)
   570→  );
   571→  const unsourcedConsumables = Array.from(allConsumed).filter(
   572→    (r) => !allProduced.has(r)
   573→  );
   574→  const resourceSufficiency = {
   575→    has_dead_resources: deadResources.length > 0,
   576→    has_unsourced_consumables: unsourcedConsumables.length > 0,
   577→    dead_resources: deadResources,
   578→    unsourced_consumables: unsourcedConsumables,
   579→  };
   580→
   581→  // Checklist: 5 criteria
   582→  const checklistItems = [
   583→    funCheckPassed,
   584→    loopClosedness.is_closed,
   585→    !resourceSufficiency.has_dead_resources && !resourceSufficiency.has_unsourced_consumables,
   586→    pathologies.critical_count === 0,
   587→    steps.length >= 3 && steps.length <= 7,
   588→  ];
   589→  const checklistPassed = checklistItems.filter(Boolean).length;
   590→  const checklistTotal = 5;
   591→
   592→  const score = Number((checklistPassed / checklistTotal).toFixed(3));
   593→  const overallPassed = checklistPassed >= 4;
   594→
   595→  const warnings: string[] = [];
   596→  if (resourceSufficiency.has_dead_resources)
   597→    warnings.push(`Dead resources: ${deadResources.join(", ")}`);
   598→  if (resourceSufficiency.has_unsourced_consumables)
   599→    warnings.push(`Unsourced consumables: ${unsourcedConsumables.join(", ")}`);
   600→  if (pathologies.critical_count > 0)
   601→    warnings.push(`${pathologies.critical_count} critical pathologies detected`);
   602→  if (!structuralType.has_braking)
   603→    warnings.push("Loop has no braking — consider adding a balancing step");
   604→
   605→  return {
   606→    fun_check: {
   607→      passed: funCheckPassed,
   608→      score: Number(funCheckScore.toFixed(3)),
   609→      reasoning: `${positiveCount}/${steps.length} positive-feedback steps; ${steps.length} steps total`,
   610→    },
   611→    loop_closedness: loopClosedness,
   612→    resource_sufficiency: resourceSufficiency,
   613→    checklist_passed: checklistPassed,
   614→    checklist_total: checklistTotal,
   615→    overall_passed: overallPassed,
   616→    score,
   617→    warnings,
   618→  };
   619→}

   621→function buildRecommendations(
   622→  pathologies: {
   623→    pathologies: Array<{
   624→      name: string;
   625→      type: string;
   626→      severity: string;
   627→      correction: string;
   628→    }>;
   629→  },
   630→  structuralType: { type: string; has_braking: boolean }

   633→    target: string;
   634→    recommendation: string;
   635→    priority: string;
   636→    category: string;
   637→    source: string;
   638→  }> = [];
   639→
   640→  for (const p of pathologies.pathologies) {
   641→    recommendations.push({
   642→      target: `Fix ${p.name}`,
   643→      recommendation: p.correction,
   644→      priority: p.severity === "critical" ? "high" : p.severity === "warning" ? "medium" : "low",
   645→      category: p.type,
   646→      source: "formal",
   647→    });
   648→  }
   649→
   650→  if (!structuralType.has_braking) {
   651→    recommendations.push({
   652→      target: "Add braking mechanism",
   653→      recommendation: "Insert a sink step that drains accumulated resources to prevent runaway",
   654→      priority: "medium",
   655→      category: "balancing",
   656→      source: "formal",
   657→    });
   658→  }
   659→
   660→  // === Type-specific recommendations for new core loop types ===
   661→
   662→  const loopType = structuralType.type;
   663→
   664→  if (loopType === "tower_defense") {
   665→    recommendations.push({
   666→      target: "Wave pacing curve",
   667→      recommendation: "Design wave intervals as 15s → 12s → 10s to create rising tension without overwhelming the player early",
   668→      priority: "high",
   669→      category: "pacing",
   670→      source: "formal",
   671→    });
   672→    recommendations.push({
   673→      target: "Economy vs defense tension",
   674→      recommendation: "Force a build-vs-defend decision every wave: spending on economy now means weaker defense next wave",
   675→      priority: "medium",
   676→      category: "decision_design",
   677→      source: "formal",
   678→    });
   679→    recommendations.push({
   680→      target: "Tower upgrade path",
   681→      recommendation: "Add at least 2 upgrade tiers per tower type (damage + range) to give long-term progression",
   682→      priority: "medium",
   683→      category: "progression",
   684→      source: "formal",
   685→    });
   686→  }
   687→
   688→  if (loopType === "rhythm") {
   689→    recommendations.push({
   690→      target: "Difficulty ramp curve",
   691→      recommendation: "Start at 60 BPM, increase by 10-15 BPM every 30 seconds — let players warm up before peak tempo",
   692→      priority: "high",
   693→      category: "pacing",
   694→      source: "formal",
   695→    });
   696→    recommendations.push({
   697→      target: "Visual feedback sync",
   698→      recommendation: "Ensure note visuals pulse exactly on beat — even 50ms visual delay breaks the rhythm feel",
   699→      priority: "high",
   700→      category: "feedback",
   701→      source: "formal",
   702→    });
   703→    recommendations.push({
   704→      target: "Miss recovery window",
   705→      recommendation: "Add a 200ms grace window after a miss where combo doesn't reset — reduces frustration without removing challenge",
   706→      priority: "medium",
   707→      category: "forgiveness",
   708→      source: "formal",
   709→    });
   710→  }
   711→
   712→  if (loopType === "puzzle") {
   713→    recommendations.push({
   714→      target: "Piece preview queue",
   715→      recommendation: "Show the next 3 pieces in a preview panel — lets players plan strategy instead of reacting randomly",
   716→      priority: "high",
   717→      category: "strategic_depth",
   718→      source: "formal",
   719→    });
   720→    recommendations.push({
   721→      target: "Hold/swap mechanic",
   722→      recommendation: "Allow players to hold one piece and swap it later — adds tactical layer and recovery option",
   723→      priority: "medium",
   724→      category: "mechanic_depth",
   725→      source: "formal",
   726→    });
   727→    recommendations.push({
   728→      target: "Difficulty via speed, not complexity",
   729→      recommendation: "Increase fall speed gradually (not piece variety) — cognitive overload kills fun faster than speed",
   730→      priority: "medium",
   731→      category: "difficulty_curve",
   732→      source: "formal",
   733→    });
   734→  }
   735→
   736→  // AI-style recommendation
   737→  recommendations.push({
   738→    target: "Playtest the 30-second fun test",
   739→    recommendation: "Run a paper prototype and time the first 30 seconds — does the player smile?",
   740→    priority: "low",
   741→    category: "validation",
   742→    source: "ai",
   743→  });
   744→
   745→  return recommendations;

   754→export async function POST(request: NextRequest) {
   755→  const startedAt = Date.now();
   756→  const user = await getCurrentUser(request);
   757→  if (!user) return UNAUTH();
   758→
   759→  try {
   760→    const body = await request.json().catch(() => ({}));
   761→    const projectId = body?.project_id?.toString().trim() || undefined;
   762→    const useAi = body?.use_ai === true || body?.use_ai === "true";
   763→    const conceptId = body?.concept_id?.toString().trim() || "standalone";
   764→    const genre = body?.genre?.toString().trim() || "action";
   765→    const desiredLoopType = body?.desired_loop_type?.toString().trim() || undefined;
   766→
   767→    const mechanics: string[] = Array.isArray(body?.mechanics)
   768→      ? body.mechanics.map((m: unknown) => String(m).trim()).filter(Boolean)
   769→      : [];
   770→    const customSteps: string[] | undefined = Array.isArray(body?.custom_steps)
   771→      ? body.custom_steps.map((s: unknown) => String(s).trim()).filter(Boolean)
   772→      : undefined;
   773→
   774→    if (mechanics.length < 1) {
   775→      return VALIDATION_ERROR(
   776→        "Поле 'mechanics' обязательно и должно содержать хотя бы одну механику"
   777→      );
   778→    }
   779→    if (desiredLoopType && !VALID_LOOP_TYPES.includes(desiredLoopType)) {
   780→      return VALIDATION_ERROR(
   781→        `Неверный desired_loop_type: ${desiredLoopType}. Допустимо: ${VALID_LOOP_TYPES.join(", ")}`
   782→      );
   783→    }
   784→
   785→    // --- Resolve project ---
   786→    const owned = await getOwnedProject(user, projectId);
   787→    if (owned instanceof NextResponse) return owned;
   788→    const proj = owned.project as {
   789→      id: string;
   790→      name: string;
   791→      genre: string | null;
   792→      concept?: { onePagerData?: string | null } | null;
   793→    };
   794→
   795→    // --- Stage 1: Classify structural type ---
   796→    // Build steps first (needed for classification)
   797→    const steps = buildSteps(
   798→      mechanics,
   799→      customSteps,
   800→      desiredLoopType || GENRE_DEFAULT_LOOP_TYPE[genre] || "hybrid"
   801→    );
   802→
   803→    const structuralType = classifyStructuralType(
   804→      mechanics,
   805→      genre,
   806→      desiredLoopType,
   807→      steps
   808→    );
   809→
   810→    // --- Stage 2: Build loop hierarchy ---
   811→    const loopHierarchy = buildLoopHierarchy(steps, structuralType.type);
   812→
   813→    // Inner loops (small timescale)
   814→    const innerLoops = [
   815→      {
   816→        name: "micro_action_loop",
   817→        actions: steps.slice(0, 2).map((s) => s.action),
   818→        duration_estimate: steps
   819→          .slice(0, 2)
   820→          .reduce((s, st) => s + st.duration_estimate, 0),
   821→        type: structuralType.type,
   822→      },
   823→    ];
   824→    // Outer loops (medium/large timescale)
   825→    const outerLoops = [
   826→      {
   827→        name: "session_loop",
   828→        actions: ["Complete 3 core loops", "Bank progress", "Trigger event"],
   829→        duration_estimate: 300,
   830→        type: "outer",
   831→      },
   832→    ];
   833→    // Meta loop (weeks/months)
   834→    const metaLoop = {
   835→      name: "meta_progression",
   836→      actions: ["New Game+", "Season pass", "Daily challenges"],
   837→      duration_estimate: 604800, // 1 week in seconds
   838→      type: "meta",
   839→    };
   840→
   841→    // --- Stage 3: Detect pathologies ---
   842→    const pathologies = detectPathologies(steps, structuralType);
   843→
   844→    // --- Stage 4: Validation ---
   845→    const validation = buildValidation(steps, pathologies, structuralType);
   846→
   847→    // --- Stage 5: Recommendations ---
   848→    const recommendations = buildRecommendations(pathologies, structuralType);
   849→
   850→    const latencyMs = Date.now() - startedAt;
   851→    const stagesCompleted = [1, 2, 3, 4, 5];
   852→
   853→    const result = {
   854→      id: proj.id,
   855→      structural_type: structuralType,
   856→      steps,
   857→      inner_loops: innerLoops,
   858→      outer_loops: outerLoops,
   859→      meta_loop: metaLoop,
   860→      pathologies,
   861→      recommendations,
   862→      validation,
   863→      loop_hierarchy: loopHierarchy,
   864→      stages_completed: stagesCompleted,
   865→      latency_ms: latencyMs,
   866→      models_used: ["deterministic-coreloop-v1", "sellers-typology", "pathology-detector-v1"],
   867→    };
   868→
   869→    // --- Optional AI enrichment ---
   870→    let aiInsights: string | null = null;
   871→    if (useAi) {
   872→      aiInsights = await enrichCoreLoop({
   873→        projectName: proj.name || "Untitled",
   874→        genre,
   875→        coreLoopType: structuralType.type,
   876→        steps: steps.map((s) => s.action),
   877→      });
   878→      if (aiInsights) {
   879→        result.ai_insights = aiInsights;
   880→        result.models_used.push("glm-4.6 (ai-enrichment)");
   881→      }
   882→    }
   883→
   884→    // --- Persist ---
   885→    const inputData = JSON.stringify({
   886→      concept_id: conceptId,
   887→      mechanics,
   888→      genre,
   889→      desired_loop_type: desiredLoopType,
   890→      custom_steps: customSteps,
   891→    });
   892→
   893→    const stepsData = JSON.stringify(steps);
   894→    const fullProfile = JSON.stringify(result);
   895→
   896→    await db.projectCoreLoop.upsert({
   897→      where: { projectId: proj.id },
   898→      create: {
   899→        projectId: proj.id,
   900→        structuralType: structuralType.type,
   901→        structuralSubtype: structuralType.sub_type,
   902→        stepCount: steps.length,
   903→        hierarchyDepth: 6, // micro → meta
   904→        pathologyCount: pathologies.total_count,
   905→        inputData,
   906→        stepsData,
   907→        innerLoops: JSON.stringify(innerLoops),
   908→        outerLoops: JSON.stringify(outerLoops),
   909→        metaLoop: JSON.stringify(metaLoop),
   910→        loopHierarchy: JSON.stringify(loopHierarchy),
   911→        pathologies: JSON.stringify(pathologies),
   912→        recommendations: JSON.stringify(recommendations),
   913→        validationData: JSON.stringify(validation),
   914→        fullProfile,
   915→      },
   916→      update: {
   917→        structuralType: structuralType.type,
   918→        structuralSubtype: structuralType.sub_type,
   919→        stepCount: steps.length,
   920→        hierarchyDepth: 6,
   921→        pathologyCount: pathologies.total_count,
   922→        inputData,
   923→        stepsData,
   924→        innerLoops: JSON.stringify(innerLoops),
   925→        outerLoops: JSON.stringify(outerLoops),
   926→        metaLoop: JSON.stringify(metaLoop),
   927→        loopHierarchy: JSON.stringify(loopHierarchy),
   928→        pathologies: JSON.stringify(pathologies),
   929→        recommendations: JSON.stringify(recommendations),
   930→        validationData: JSON.stringify(validation),
   931→        fullProfile,
   932→      },
   933→    });
   934→
   935→    await updateProjectStage(proj.id, "core_loop");
   936→
   937→    // Use safeJsonParse to satisfy linter (kept for future use)
   938→    void safeJsonParse;
   939→
   940→    return NextResponse.json(result);
   941→  } catch (error) {
   942→    console.error("[coreloop/design] error:", error);
   943→    return SERVER_ERROR();
   944→  }
   945→}
