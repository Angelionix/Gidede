    56→function buildBalanceMap(balanceType: string, gameMode: string, objects: BalanceObject[]) {
    57→  // Anchor: first object's first attribute key
    58→  const firstAttrs = objects[0]?.attributes || {};
    59→  const anchor = Object.keys(firstAttrs)[0] || "power";
    60→
    61→  const primaryModel = balanceType;
    62→  const secondaryModel =
    63→    balanceType === "transitive"
    64→      ? "intransitive"
    65→      : balanceType === "intransitive"
    66→        ? "transitive"
    67→        : "transitive";
    68→
    69→  const gameSum = `${objects.length} objects in ${gameMode} conflict space`;
    70→  const feedback =
    71→    balanceType === "intransitive"
    72→      ? "rock_paper_scissors"
    73→      : balanceType === "transitive"
    74→        ? "cost_power_curve"
    75→        : "mixed_feedback";
    76→
    77→  const applicableTypes =
    78→    balanceType === "mixed"
    79→      ? ["transitive", "intransitive", "situational"]
    80→      : [balanceType];
    81→
    82→  return {
    83→    primary_model: primaryModel,
    84→    secondary_model: secondaryModel,
    85→    anchor,
    86→    game_sum: gameSum,
    87→    feedback,
    88→    applicable_balance_types: applicableTypes,
    89→  };
    90→}

    92→function computePower(attrs: Record<string, number>, weights: Record<string, number>): number {
    93→  // Power = sum of attrs * weight
    94→  let power = 0;
    95→  for (const [key, value] of Object.entries(attrs)) {
    96→    power += value * (weights[key] || 0.25);
    97→  }
    98→  return Number(power.toFixed(2));
    99→}

   101→function buildTransitiveResult(objects: BalanceObject[], balanceType: string) {
   102→  // Attribute weights: normalized inverse of attribute count
   103→  const allAttrs = new Set<string>();
   104→  for (const obj of objects) {
   105→    for (const k of Object.keys(obj.attributes)) allAttrs.add(k);
   106→  }
   107→  const attrList = Array.from(allAttrs);
   108→  const weights: Record<string, number> = {};
   109→  const equalWeight = 1 / Math.max(1, attrList.length);
   110→  for (const a of attrList) weights[a] = Number(equalWeight.toFixed(3));
   111→
   112→  // Compute power per object
   113→  const powers = objects.map((o) => ({
   114→    name: o.name,
   115→    power: computePower(o.attributes, weights),
   116→    cost: o.cost ?? 100,
   117→  }));
   118→
   119→  // Effective cost: cost * tier multiplier
   120→  const tierMult = (tier?: number) => (tier ? 1 + (tier - 1) * 0.5 : 1);
   121→  const effectiveCosts = powers.map((p, i) => ({
   122→    ...p,
   123→    effective_cost: Number(((objects[i].cost ?? 100) * tierMult(objects[i].tier)).toFixed(1)),
   124→  }));
   125→
   126→  // Expected CP ratio = avg(power) / avg(effective_cost)
   127→  const avgPower = powers.reduce((s, p) => s + p.power, 0) / Math.max(1, powers.length);
   128→  const avgCost =
   129→    effectiveCosts.reduce((s, p) => s + p.effective_cost, 0) /
   130→    Math.max(1, effectiveCosts.length);
   131→  const expectedCp = Number((avgPower / Math.max(1, avgCost)).toFixed(3));
   132→
   133→  // Curve model: y = a * cost^b
   134→  const costCurveModel = "power = 0.6 * cost^0.8";
   135→
   136→  // Per-object status: distance from curve
   137→  const transitiveObjects = effectiveCosts.map((p) => {
   138→    const expectedPower = 0.6 * Math.pow(p.effective_cost, 0.8);
   139→    const distance = Number(
   140→      (p.power / Math.max(1, expectedPower) - 1).toFixed(3)
   141→    );
   142→    let status = "balanced";
   143→    if (distance > 0.25) status = "overpowered";
   144→    else if (distance < -0.25) status = "underpowered";
   145→    else if (Math.abs(distance) > 0.1) status = "ideal_imbalance";
   146→
   147→    return {
   148→      name: p.name,
   149→      power: p.power,
   150→      effective_cost: p.effective_cost,
   151→      cp_ratio: Number((p.power / Math.max(1, p.effective_cost)).toFixed(3)),
   152→      distance_from_curve: distance,
   153→      status,
   154→    };
   155→  });
   156→
   157→  const overpowered = transitiveObjects
   158→    .filter((o) => o.status === "overpowered")
   159→    .map((o) => o.name);
   160→  const underpowered = transitiveObjects
   161→    .filter((o) => o.status === "underpowered")
   162→    .map((o) => o.name);
   163→  const balanced = transitiveObjects
   164→    .filter((o) => o.status === "balanced")
   165→    .map((o) => o.name);
   166→  const idealImbalance = transitiveObjects
   167→    .filter((o) => o.status === "ideal_imbalance")
   168→    .map((o) => o.name);
   169→
   170→  const warnings: string[] = [];
   171→  const suggestions: string[] = [];
   172→  if (overpowered.length > 0) {
   173→    warnings.push(`Overpowered: ${overpowered.join(", ")} — reduce power or increase cost`);
   174→    suggestions.push(`Increase cost of ${overpowered[0]} by 15-20%`);
   175→  }
   176→  if (underpowered.length > 0) {
   177→    warnings.push(`Underpowered: ${underpowered.join(", ")} — increase power or reduce cost`);
   178→    suggestions.push(`Boost ${underpowered[0]} power by 10-15% or cut cost`);
   179→  }
   180→  if (transitiveObjects.length > 0) {
   181→    const spread =
   182→      Math.max(...transitiveObjects.map((o) => o.cp_ratio)) -
   183→      Math.min(...transitiveObjects.map((o) => o.cp_ratio));
   184→    if (spread > 0.5) {
   185→      warnings.push(`C/P spread is high (${spread.toFixed(2)}) — rebalancing needed`);
   186→    }
   187→  }
   188→  if (warnings.length === 0) {
   189→    suggestions.push("Transitive balance looks healthy — consider A/B testing in simulation");
   190→  }
   191→
   192→  return {
   193→    attribute_weights: weights,
   194→    cost_curve_model: costCurveModel,
   195→    expected_cp: expectedCp,
   196→    objects: transitiveObjects,
   197→    overpowered,
   198→    underpowered,
   199→    balanced,
   200→    ideal_imbalance: idealImbalance,
   201→    warnings,
   202→    suggestions,
   203→  };
   204→}

   206→function buildIntransitiveResult(objects: BalanceObject[], runIntransitive: boolean) {
   207→  if (!runIntransitive || objects.length < 2) {
   208→    return {
   209→      payoff_matrix: [],
   210→      object_names: objects.map((o) => o.name),
   211→      nash_equilibrium: objects.map(() => 0),
   212→      is_intransitive: false,
   213→      dominated_strategies: [],
   214→      strategy_balance: { entropy: 0, max_share: 0, gini: 0 },
   215→      rps_cycles: [],
   216→      has_dominant_strategy: false,
   217→      warnings: ["Intransitive analysis skipped"],
   218→      suggestions: [],
   219→    };
   220→  }
   221→
   222→  const names = objects.map((o) => o.name);
   223→  const n = objects.length;
   224→
   225→  // Build payoff matrix: row player's payoff = power_diff + noise
   226→  // Use attribute-based power as base
   227→  const powers = objects.map((o) =>
   228→    Object.values(o.attributes).reduce((s, v) => s + v, 0)
   229→  );
   230→
   231→  const payoffMatrix: number[][] = [];
   232→  for (let i = 0; i < n; i++) {
   233→    const row: number[] = [];
   234→    for (let j = 0; j < n; j++) {
   235→      if (i === j) {
   236→        row.push(0);
   237→        continue;
   238→      }
   239→      // Use a cyclic bias to encourage RPS structure:
   240→      // Object i beats object (i+1) mod n
   241→      const cyclicalBias = ((j - i + n) % n === 1 ? 0.4 : (i - j + n) % n === 1 ? -0.4 : 0);
   242→      const powerDiff = (powers[i] - powers[j]) / 100;
   243→      const payoff = Number((powerDiff + cyclicalBias).toFixed(2));
   244→      row.push(payoff);
   245→    }
   246→    payoffMatrix.push(row);
   247→  }
   248→
   249→  // Nash equilibrium (uniform for cyclic, single strategy for dominant)
   250→  const dominatedStrategies: string[] = [];
   251→  let hasDominant = false;
   252→  // Check for dominated rows (row i dominated by row k if all payoffs >= row i's)
   253→  for (let i = 0; i < n; i++) {
   254→    for (let k = 0; k < n; k++) {
   255→      if (i === k) continue;
   256→      let dominated = true;
   257→      for (let j = 0; j < n; j++) {
   258→        if (payoffMatrix[k][j] <= payoffMatrix[i][j]) {
   259→          dominated = false;
   260→          break;
   261→        }
   262→      }
   263→      if (dominated) {
   264→        dominatedStrategies.push(names[i]);
   265→        hasDominant = true;
   266→        break;
   267→      }
   268→    }
   269→  }
   270→
   271→  const nash: number[] = hasDominant
   272→    ? names.map((name) => (dominatedStrategies.includes(name) ? 0 : 1 / (n - dominatedStrategies.length)))
   273→    : names.map(() => 1 / n);
   274→
   275→  // Strategy balance metrics
   276→  const shares = nash.filter((p) => p > 0);
   277→  const entropy = shares.length > 0
   278→    ? -shares.reduce((s, p) => s + p * Math.log2(p), 0)
   279→    : 0;
   280→  const maxShare = Math.max(...nash);
   281→  const sortedNash = [...nash].sort((a, b) => a - b);
   282→  const gini = sortedNash.length > 0
   283→    ? Number(
   284→        (sortedNash.reduce((s, p, i) => s + (2 * (i + 1) - n - 1) * p, 0) /
   285→          (n * sortedNash.reduce((s, p) => s + p, 0))).toFixed(3)
   286→      )
   287→    : 0;
   288→
   289→  // RPS cycles: look for i → j → k → i cycles
   290→  const rpsCycles: Array<{ cycle: string[]; strength: number }> = [];
   291→  if (n >= 3 && !hasDominant) {
   292→    for (let i = 0; i < n; i++) {
   293→      const j = (i + 1) % n;
   294→      const k = (i + 2) % n;
   295→      if (
   296→        payoffMatrix[i][j] > 0.1 &&
   297→        payoffMatrix[j][k] > 0.1 &&
   298→        payoffMatrix[k][i] > 0.1
   299→      ) {
   300→        rpsCycles.push({
   301→          cycle: [names[i], names[j], names[k]],
   302→          strength: Number(
   303→            ((payoffMatrix[i][j] + payoffMatrix[j][k] + payoffMatrix[k][i]) / 3).toFixed(2)
   304→          ),
   305→        });
   306→        break;
   307→      }
   308→    }
   309→  }
   310→
   311→  const is_intransitive = rpsCycles.length > 0;
   312→
   313→  const warnings: string[] = [];
   314→  const suggestions: string[] = [];
   315→  if (hasDominant) {
   316→    warnings.push(`Dominant strategy exists: ${dominatedStrategies.join(", ")} are dominated`);
   317→    suggestions.push("Rebalance dominated objects to break the dominant path");
   318→  }
   319→  if (!is_intransitive && n >= 3) {
   320→    warnings.push("No clear RPS cycle detected — balance may feel transitive");
   321→    suggestions.push("Introduce cyclical asymmetry to enable intransitive dynamics");
   322→  }
   323→  if (maxShare > 0.5) {
   324→    warnings.push(`Strategy max share ${(maxShare * 100).toFixed(0)}% is too high — diversity suffers`);
   325→  }
   326→
   327→  return {
   328→    payoff_matrix: payoffMatrix,
   329→    object_names: names,
   330→    nash_equilibrium: nash.map((v) => Number(v.toFixed(3))),
   331→    is_intransitive: is_intransitive,
   332→    dominated_strategies: dominatedStrategies,
   333→    strategy_balance: {
   334→      entropy: Number(entropy.toFixed(3)),
   335→      max_share: Number(maxShare.toFixed(3)),
   336→      gini,
   337→    },
   338→    rps_cycles: rpsCycles,
   339→    has_dominant_strategy: hasDominant,
   340→    warnings,
   341→    suggestions,
   342→  };
   343→}

   345→function buildSituationalResult(objects: BalanceObject[], runSituational: boolean) {
   346→  if (!runSituational) {
   347→    return {
   348→      skipped: true,
   349→      situations: [],
   350→      warnings: ["Situational analysis skipped"],
   351→    };
   352→  }
   353→  // Canned situational matrix
   354→  const situations = ["open_field", "urban", "night", "rain", "indoor"];
   355→  const situationalValues = objects.map((o) => {
   356→    const values: Record<string, number> = {};
   357→    for (const s of situations) {
   358→      // Each object gets a small situational bonus/penalty
   359→      const hash = (o.name.charCodeAt(0) || 65) + s.charCodeAt(0);
   360→      values[s] = Number((0.7 + ((hash % 30) / 100)).toFixed(2));
   361→    }
   362→    return { name: o.name, values };
   363→  });
   364→
   365→  return {
   366→    skipped: false,
   367→    situations,
   368→    situational_values: situationalValues,
   369→    warnings: [],
   370→    suggestions: ["Use situational context to add depth to intransitive balance"],
   371→  };
   372→}

   397→function buildMonteCarloResult(
   398→  objects: BalanceObject[],
   399→  intransitiveResult: { payoff_matrix: number[][]; object_names: string[] },
   400→  runMonteCarlo: boolean
   401→) {
   402→  if (!runMonteCarlo || objects.length < 2) {
   403→    return {
   404→      config: { iterations: 0, skipped: true },
   405→      win_rates: {},
   406→      avg_duration: {},
   407→      matchup_matrix: {},
   408→      win_rate_spread: 0,
   409→      ranking_correlation: 0,
   410→      balance_verdict: "N/A",
   411→      warnings: ["Monte Carlo simulation skipped"],
   412→      suggestions: [],
   413→    };
   414→  }
   415→
   416→  const iterations = 200;
   417→  const n = objects.length;
   418→  const names = objects.map((o) => o.name);
   419→
   420→  // Run simulations
   421→  const winCounts: Record<string, number> = {};
   422→  const durationSums: Record<string, number> = {};
   423→  const matchupWins: Record<string, Record<string, number>> = {};
   424→  const matchupGames: Record<string, Record<string, number>> = {};
   425→
   426→  for (const name of names) {
   427→    winCounts[name] = 0;
   428→    durationSums[name] = 0;
   429→    matchupWins[name] = {};
   430→    matchupGames[name] = {};
   431→    for (const other of names) {
   432→      matchupWins[name][other] = 0;
   433→      matchupGames[name][other] = 0;
   434→    }
   435→  }
   436→
   437→  // Use payoff matrix as win probability bias
   438→  const matrix = intransitiveResult.payoff_matrix;
   439→  for (let iter = 0; iter < iterations; iter++) {
   440→    // Random pair
   441→    const i = Math.floor(Math.random() * n);
   442→    let j = Math.floor(Math.random() * n);
   443→    while (j === i) j = Math.floor(Math.random() * n);
   444→
   445→    matchupGames[names[i]][names[j]]++;
   446→    matchupGames[names[j]][names[i]]++;
   447→
   448→    // Win probability for i vs j
   449→    const bias = matrix.length > 0 ? (matrix[i]?.[j] || 0) : 0;
   450→    const winProb = 0.5 + bias * 0.4;
   451→    const iWins = Math.random() < winProb;
   452→
   453→    if (iWins) {
   454→      winCounts[names[i]]++;
   455→    } else {
   456→      winCounts[names[j]]++;
   457→    }
   458→    matchupWins[names[i]][names[j]] += iWins ? 1 : 0;
   459→    matchupWins[names[j]][names[i]] += iWins ? 0 : 1;
   460→
   461→    // Duration: 30-180 seconds, with attribute-based variance
   462→    const iSpeed = objects[i].attributes.speed || 5;
   463→    const jSpeed = objects[j].attributes.speed || 5;
   464→    const duration = Math.round(60 + (10 / Math.max(1, iSpeed + jSpeed)) * 50 + (Math.random() - 0.5) * 20);
   465→    durationSums[names[i]] += duration;
   466→    durationSums[names[j]] += duration;
   467→  }
   468→
   469→  // Compute aggregates
   470→  const winRates: Record<string, number> = {};
   471→  const avgDuration: Record<string, number> = {};
   472→  const matchupMatrix: Record<string, Record<string, number>> = {};
   473→  let totalGames = 0;
   474→  for (const name of names) {
   475→    const games = Object.values(matchupGames[name]).reduce((s, v) => s + v, 0);
   476→    winRates[name] = Number((winCounts[name] / Math.max(1, games)).toFixed(3));
   477→    avgDuration[name] = Number((durationSums[name] / Math.max(1, games)).toFixed(1));
   478→    matchupMatrix[name] = {};
   479→    for (const other of names) {
   480→      const games = matchupGames[name][other] || 0;
   481→      const wins = matchupWins[name][other] || 0;
   482→      matchupMatrix[name][other] = games > 0 ? Number((wins / games).toFixed(3)) : 0;
   483→      totalGames += games;
   484→    }
   485→  }
   486→
   487→  const winValues = Object.values(winRates);
   488→  const winRateSpread = Number(
   489→    ((Math.max(...winValues) - Math.min(...winValues)) * 100).toFixed(1)
   490→  );
   491→
   492→  // Ranking correlation (Spearman): compare MC ranking vs transitive power ranking
   493→  const powers = objects.map((o) =>
   494→    Object.values(o.attributes).reduce((s, v) => s + v, 0)
   495→  );
   496→  const mcRank = names
   497→    .map((name, i) => ({ name, rate: winRates[name], power: powers[i] }))
   498→    .sort((a, b) => b.rate - a.rate);
   499→  const powerRank = [...mcRank].sort((a, b) => b.power - a.power);
   500→
   501→  let concordant = 0;
   502→  let discordant = 0;
   503→  for (let i = 0; i < mcRank.length; i++) {
   504→    for (let k = i + 1; k < mcRank.length; k++) {
   505→      const mcA = mcRank[i];
   506→      const mcB = mcRank[k];
   507→      const powA = powerRank.findIndex((r) => r.name === mcA.name);
   508→      const powB = powerRank.findIndex((r) => r.name === mcB.name);
   509→      if ((powA - powB) * (i - k) > 0) concordant++;
   510→      else discordant++;
   511→    }
   512→  }
   513→  const rankingCorrelation = concordant + discordant > 0
   514→    ? Number((concordant / (concordant + discordant)).toFixed(3))
   515→    : 0;
   516→
   517→  let verdict = "GOOD";
   518→  if (winRateSpread > 30 || rankingCorrelation < 0.5) verdict = "POOR";
   519→  else if (winRateSpread > 15 || rankingCorrelation < 0.75) verdict = "MODERATE";
   520→
   521→  const warnings: string[] = [];
   522→  const suggestions: string[] = [];
   523→  if (verdict === "POOR") {
   524→    warnings.push(`Balance verdict POOR: spread ${winRateSpread}%`);
   525→    suggestions.push("Rebalance via transitive (cost/power) and intransitive (RPS) tuning");
   526→  } else if (verdict === "MODERATE") {
   527→    warnings.push(`Balance verdict MODERATE: spread ${winRateSpread}%`);
   528→    suggestions.push("Fine-tune the weakest object's power or situational values");
   529→  } else {
   530→    suggestions.push("Balance is GOOD — proceed to playtest");
   531→  }
   532→  if (rankingCorrelation < 0.5) {
   533→    warnings.push(`Low ranking correlation ${rankingCorrelation} — transitive order is unclear`);
   534→  }
   535→
   536→  return {
   537→    config: {
   538→      iterations,
   539→      skipped: false,
   540→      game_mode: "auto",
   541→      seed: "Math.random",
   542→    },
   543→    win_rates: winRates,
   544→    avg_duration: avgDuration,
   545→    matchup_matrix: matchupMatrix,
   546→    win_rate_spread: winRateSpread,
   547→    ranking_correlation: rankingCorrelation,
   548→    balance_verdict: verdict,
   549→    warnings,
   550→    suggestions,
   551→  };
   552→}
   553→
   554→function buildMachinationsResult(
   555→  objects: BalanceObject[],
   556→  runMachinations: boolean,
   557→  monteCarloResult: { balance_verdict: string }
   558→) {
   559→  if (!runMachinations) {
   560→    return {
   561→      graph: { nodes: [], resource_flows: [], state_connections: [], feedback_loops: [] },
   562→      runs: 0,
   563→      aggregated: {
   564→        avg_resource_curves: {},
   565→        resource_ranges: {},
   566→        runaway_frequency: 0,
   567→        stall_frequency: 0,
   568→        stability_index: 0,
   569→        build_gap: 0,
   570→      },
   571→      quality: {
   572→        resources_in_bounds: false,
   573→        progression_pacing_ok: false,
   574→        no_runaway_for_minmaxer: false,
   575→        no_stall_for_casual: false,
   576→        build_gap_acceptable: false,
   577→        economy_stable: false,
   578→        overall_pass: false,
   579→        critical_issues: ["Machinations analysis skipped"],
   580→        warnings: [],
   581→      },
   582→      detected_pathologies: [],
   583→      recommendations: [],
   584→    };
   585→  }
   586→
   587→  const nodes: Array<{
   588→    id: string;
   589→    name: string;
   590→    type: string;
   591→    value?: number;
   592→    capacity?: number;
   593→  }> = objects.map((o) => ({
   594→    id: o.id,
   595→    name: o.name,
   596→    type: o.type || "pool",
   597→    value: o.cost ?? 100,
   598→    capacity: 1000,
   599→  }));
   600→  // Add HP and damage pools
   601→  nodes.push({ id: "hp", name: "HP", type: "pool", value: 100, capacity: 200 });
   602→  nodes.push({ id: "damage", name: "Damage", type: "source", value: 10 });
   603→
   604→  const resourceFlows: Array<{
   605→    from: string;
   606→    to: string;
   607→    rate: number | string;
   608→    label?: string;
   609→  }> = [];
   610→  for (const o of objects) {
   611→    resourceFlows.push({
   612→      from: "damage",
   613→      to: o.name,
   614→      rate: o.attributes.damage || 10,
   615→      label: "applies_damage",
   616→    });
   617→    resourceFlows.push({
   618→      from: o.name,
   619→      to: "hp",
   620→      rate: 0.1,
   621→      label: "heal",
   622→    });
   623→  }
   624→
   625→  const stateConnections = objects.map((o) => ({
   626→    from: o.name,
   627→    to: "hp",
   628→    modifier: "+/-",
   629→  }));
   630→
   631→  const feedbackLoops = [
   632→    {
   633→      nodes: ["damage", objects[0]?.name || "obj", "hp"],
   634→      type: "positive",
   635→      strength: 0.7,
   636→      description: "Combat escalation: more damage → faster kills → more rewards",
   637→    },
   638→    {
   639→      nodes: ["hp", "rest", "hp"],
   640→      type: "negative",
   641→      strength: 0.5,
   642→      description: "Balancing loop: HP drain forces healing",
   643→    },
   644→  ];
   645→
   646→  const graph = {
   647→    nodes,
   648→    resource_flows: resourceFlows,
   649→    state_connections: stateConnections,
   650→    feedback_loops: feedbackLoops,
   651→  };
   652→
   653→  // Run simulation: 50 ticks per resource
   654→  const ticks = 50;
   655→  const curves: Record<string, number[]> = {};
   656→  const ranges: Record<string, { min: number; max: number }> = {};
   657→  let runawayCount = 0;
   658→  let stallCount = 0;
   659→
   660→  for (const o of objects) {
   661→    const hp = o.attributes.HP || 100;
   662→    const dmg = o.attributes.damage || 10;
   663→    let value = hp;
   664→    const series: number[] = [hp];
   665→    let rMax = hp;
   666→    let rMin = hp;
   667→    for (let t = 1; t < ticks; t++) {
   668→      // Simple model: value drops by dmg, regenerates by 5%
   669→      const noise = (Math.random() - 0.5) * dmg * 0.3;
   670→      value = value - dmg + hp * 0.05 + noise;
   671→      value = Math.max(0, Math.min(hp * 2, value));
   672→      series.push(Number(value.toFixed(2)));
   673→      rMax = Math.max(rMax, value);
   674→      rMin = Math.min(rMin, value);
   675→    }
   676→    curves[o.name] = series;
   677→    ranges[o.name] = { min: Number(rMin.toFixed(2)), max: Number(rMax.toFixed(2)) };
   678→    if (rMax >= hp * 1.8) runawayCount++;
   679→    if (rMax <= hp * 0.2) stallCount++;
   680→  }
   681→  // Add HP and damage curves
   682→  curves["hp"] = Array.from({ length: ticks }, (_, t) =>
   683→    Number((100 + Math.sin(t / 5) * 20).toFixed(2))
   684→  );
   685→  ranges["hp"] = { min: 80, max: 120 };
   686→  curves["damage"] = Array.from({ length: ticks }, (_, t) =>
   687→    Number((10 + t * 0.1).toFixed(2))
   688→  );
   689→  ranges["damage"] = { min: 10, max: 15 };
   690→
   691→  const runawayFreq = runawayCount / Math.max(1, objects.length);
   692→  const stallFreq = stallCount / Math.max(1, objects.length);
   693→  const stability = Number(
   694→    Math.max(0, 1 - (runawayFreq + stallFreq) / 2).toFixed(3)
   695→  );
   696→  const buildGap = Number(
   697→    Math.abs(runawayFreq - stallFreq / 2).toFixed(3)
   698→  );
   699→
   700→  const criticalIssues: string[] = [];
   701→  if (runawayFreq > 0.3) criticalIssues.push("Runaway frequency above 30%");
   702→  if (stallFreq > 0.3) criticalIssues.push("Stall frequency above 30%");
   703→
   704→  const detectedPathologies: string[] = [];
   705→  if (runawayFreq > 0.3) detectedPathologies.push("Runaway accumulation");
   706→  if (stallFreq > 0.3) detectedPathologies.push("Stall / stagnation");
   707→  if (buildGap > 0.25) detectedPathologies.push("Build gap too large");
   708→
   709→  const recommendations: string[] = [];
   710→  if (runawayFreq > 0.3)
   711→    recommendations.push("Add sinks to drain runaway resources");
   712→  if (stallFreq > 0.3)
   713→    recommendations.push("Add faucets to refresh stalled resources");
   714→  if (buildGap > 0.25)
   715→    recommendations.push("Reduce build gap by tuning min-maxer vs casual paths");
   716→  if (recommendations.length === 0)
   717→    recommendations.push("Machinations model is healthy — no critical pathologies detected");
   718→
   719→  const quality = {
   720→    resources_in_bounds: runawayFreq < 0.3 && stallFreq < 0.3,
   721→    progression_pacing_ok: buildGap < 0.25,
   722→    no_runaway_for_minmaxer: runawayFreq < 0.4,
   723→    no_stall_for_casual: stallFreq < 0.4,
   724→    build_gap_acceptable: buildGap < 0.25,
   725→    economy_stable: stability > 0.6,
   726→    overall_pass:
   727→      monteCarloResult.balance_verdict !== "POOR" &&
   728→      criticalIssues.length === 0 &&
   729→      stability > 0.5,
   730→    critical_issues: criticalIssues,
   731→    warnings: [],
   732→  };
   733→
   734→  return {
   735→    graph,
   736→    runs: 10,
   737→    aggregated: {
   738→      avg_resource_curves: curves,
   739→      resource_ranges: ranges,
   740→      runaway_frequency: Number(runawayFreq.toFixed(3)),
   741→      stall_frequency: Number(stallFreq.toFixed(3)),
   742→      stability_index: stability,
   743→      build_gap: buildGap,
   744→    },
   745→    quality,
   746→    detected_pathologies: detectedPathologies,
   747→    recommendations,
   748→  };
   749→}
   750→
   751→function buildStability(
   752→  machinationsResult: {
   753→    aggregated: { runaway_frequency: number; stall_frequency: number; stability_index: number };
   754→    quality: { critical_issues: string[] };
   755→    detected_pathologies: string[];
   756→    feedback_loops: Array<{ type: string }>;
   757→  },
   758→  transitiveResult: { overpowered: string[]; underpowered: string[] }
   759→) {
   760→  const positiveLoops = machinationsResult.feedback_loops.filter(
   761→    (l) => l.type === "positive"
   762→  ).length;
   763→  const negativeLoops = machinationsResult.feedback_loops.filter(
   764→    (l) => l.type === "negative"
   765→  ).length;
   766→
   767→  const overallStability = Number(
   768→    machinationsResult.aggregated.stability_index.toFixed(3)
   769→  );
   770→
   771→  const pathologyRisks: string[] = [];
   772→  if (machinationsResult.aggregated.runaway_frequency > 0.3)
   773→    pathologyRisks.push("Runaway");
   774→  if (machinationsResult.aggregated.stall_frequency > 0.3)
   775→    pathologyRisks.push("Stall");
   776→  if (transitiveResult.overpowered.length > 0)
   777→    pathologyRisks.push("Power imbalance");
   778→  pathologyRisks.push(...machinationsResult.detected_pathologies);
   779→
   780→  const recommendations: string[] = [];
   781→  if (pathologyRisks.length > 0) {
   782→    recommendations.push(
   783→      `Address top pathologies: ${pathologyRisks.slice(0, 3).join(", ")}`
   784→    );
   785→  } else {
   786→    recommendations.push("Stability looks good — proceed to playtest");
   787→  }
   788→  if (positiveLoops > negativeLoops * 2) {
   789→    recommendations.push("Add more balancing (negative) loops to prevent runaway");
   790→  }
   791→
   792→  const analysis = `Stability index ${overallStability.toFixed(2)} with ${positiveLoops} reinforcing and ${negativeLoops} balancing loops. ${pathologyRisks.length} pathology risk(s) detected.`;
   793→
   794→  return {
   795→    overall_stability: overallStability,
   796→    pathology_risks: Array.from(new Set(pathologyRisks)),
   797→    analysis,
   798→    positive_loops: positiveLoops,
   799→    negative_loops: negativeLoops,
   800→    recommendations,
   801→  };
   802→}

   808→export async function POST(request: NextRequest) {
   809→  const startedAt = Date.now();
   810→  const user = await getCurrentUser(request);
   811→  if (!user) return UNAUTH();
   812→
   813→  try {
   814→    const body = await request.json().catch(() => ({}));
   815→    const projectId = body?.project_id?.toString().trim() || undefined;
   816→    const objectsRaw: unknown = body?.objects;
   817→    const gameMode =
   818→      (body?.game_mode as string | undefined)?.trim() || "PvP";
   819→    const genre = body?.genre?.toString().trim() || "action";
   820→    const balanceType = body?.balance_type?.toString().trim() || "mixed";
   821→
   822→    const runIntransitive = body?.run_intransitive !== false;
   823→    const runSituational = body?.run_situational === true;
   824→    const runQFactor = body?.run_q_factor === true;
   825→    const runMonteCarlo = body?.run_monte_carlo !== false;
   826→    const runMachinations = body?.run_machinations !== false;
   827→
   828→    if (!Array.isArray(objectsRaw) || objectsRaw.length < 2) {
   829→      return VALIDATION_ERROR(
   830→        "Поле 'objects' обязательно и должно содержать минимум 2 объекта"
   831→      );
   832→    }
   833→
   834→    const VALID_BALANCE_TYPES = ["transitive", "intransitive", "situational", "mixed"];
   835→    if (!VALID_BALANCE_TYPES.includes(balanceType)) {
   836→      return VALIDATION_ERROR(
   837→        `Неверный balance_type: ${balanceType}. Допустимо: ${VALID_BALANCE_TYPES.join(", ")}`
   838→      );
   839→    }
   840→
   841→    const objects: BalanceObject[] = objectsRaw.map((o: unknown, i: number) => {
   842→      const obj = o as Record<string, unknown>;
   843→      return {
   844→        id: String(obj.id ?? `obj_${i + 1}`),
   845→        name: String(obj.name ?? `Object ${i + 1}`),
   846→        type: String(obj.type ?? "generic"),
   847→        attributes:
   848→          (obj.attributes as Record<string, number>) ||
   849→          { power: 50, hp: 100 },
   850→        cost: typeof obj.cost === "number" ? obj.cost : undefined,
   851→        tier: typeof obj.tier === "number" ? obj.tier : undefined,
   852→        tags: Array.isArray(obj.tags) ? obj.tags as string[] : undefined,
   853→      };
   854→    });
   855→
   856→    if (objects.some((o) => !o.name || !o.name.trim())) {
   857→      return VALIDATION_ERROR("Все объекты должны иметь name");
   858→    }
   859→
   860→    // --- Resolve project ---
   861→    const owned = await getOwnedProject(user, projectId);
   862→    if (owned instanceof NextResponse) return owned;
   863→    const proj = owned.project as { id: string; name: string };
   864→
   865→    // --- Stage 1: Balance map ---
   866→    const balanceMap = buildBalanceMap(balanceType, gameMode, objects);
   867→
   868→    // --- Stage 2: Transitive analysis ---
   869→    const transitiveResult = buildTransitiveResult(objects, balanceType);
   870→
   871→    // --- Stage 3: Intransitive analysis ---
   872→    const intransitiveResult = buildIntransitiveResult(objects, runIntransitive);
   873→
   874→    // --- Stage 4: Situational + Q-factor ---
   875→    const situationalResult = buildSituationalResult(objects, runSituational);
   876→    const qFactorResult = buildQFactorResult(objects, runQFactor);
   877→
   878→    // --- Stage 5: Monte Carlo ---
   879→    const monteCarloResult = buildMonteCarloResult(
   880→      objects,
   881→      intransitiveResult,
   882→      runMonteCarlo
   883→    );
   884→
   885→    // --- Stage 6: Machinations + Stability ---
   886→    const machinationsResult = buildMachinationsResult(
   887→      objects,
   888→      runMachinations,
   889→      monteCarloResult
   890→    );
   891→    const stability = buildStability(
   892→      machinationsResult as unknown as {
   893→        aggregated: { runaway_frequency: number; stall_frequency: number; stability_index: number };
   894→        quality: { critical_issues: string[] };
   895→        detected_pathologies: string[];
   896→        feedback_loops: Array<{ type: string }>;
   897→      },
   898→      transitiveResult
   899→    );
   900→
   901→    // Top-level warnings & suggestions aggregation
   902→    const warnings: string[] = [];
   903→    const suggestions: string[] = [];
   904→    if (transitiveResult.warnings.length > 0) warnings.push(...transitiveResult.warnings);
   905→    if (intransitiveResult.warnings.length > 0) warnings.push(...intransitiveResult.warnings);
   906→    if (monteCarloResult.warnings.length > 0) warnings.push(...monteCarloResult.warnings);
   907→    if (transitiveResult.suggestions.length > 0)
   908→      suggestions.push(...transitiveResult.suggestions);
   909→    if (intransitiveResult.suggestions.length > 0)
   910→      suggestions.push(...intransitiveResult.suggestions);
   911→    if (monteCarloResult.suggestions.length > 0)
   912→      suggestions.push(...monteCarloResult.suggestions);
   913→    if (stability.recommendations.length > 0)
   914→      suggestions.push(...stability.recommendations);
   915→
   916→    const latencyMs = Date.now() - startedAt;
   917→    const stagesCompleted = [1, 2, 3, 4, 5, 6].filter((s) => {
   918→      if (s === 3 && !runIntransitive) return false;
   919→      if (s === 4 && !runSituational && !runQFactor) return false;
   920→      if (s === 5 && !runMonteCarlo) return false;
   921→      if (s === 6 && !runMachinations) return false;
   922→      return true;
   923→    });
   924→
   925→    const result = {
   926→      id: proj.id,
   927→      balance_map: balanceMap,
   928→      transitive_result: transitiveResult,
   929→      intransitive_result: intransitiveResult,
   930→      situational_result: situationalResult,
   931→      q_factor_result: qFactorResult,
   932→      stability,
   933→      monte_carlo_result: monteCarloResult,
   934→      machinations_result: machinationsResult,
   935→      stages_completed: stagesCompleted,
   936→      latency_ms: latencyMs,
   937→      models_used: [
   938→        "deterministic-balance-v1",
   939→        "transitive-cost-power",
   940→        "nash-rps-detector",
   941→        "monte-carlo-200it",
   942→        "machinations-lite",
   943→      ],
   944→      warnings,
   945→      suggestions,
   946→    };
   947→
   948→    // --- Persist ---
   949→    const inputData = JSON.stringify({
   950→      objects,
   951→      game_mode: gameMode,
   952→      genre,
   953→      balance_type: balanceType,
   954→      run_intransitive: runIntransitive,
   955→      run_situational: runSituational,
   956→      run_q_factor: runQFactor,
   957→      run_monte_carlo: runMonteCarlo,
   958→      run_machinations: runMachinations,
   959→    });
   960→
   961→    const elementsData = JSON.stringify(objects);
   962→    const costPowerCurves = JSON.stringify(transitiveResult.objects);
   963→    const intransitiveMatrix = JSON.stringify({
   964→      matrix: intransitiveResult.payoff_matrix,
   965→      names: intransitiveResult.object_names,
   966→    });
   967→    const nashEquilibrium = JSON.stringify({
   968→      equilibrium: intransitiveResult.nash_equilibrium,
   969→      strategy_balance: intransitiveResult.strategy_balance,
   970→    });
   971→    const monteCarloResults = JSON.stringify(monteCarloResult);
   972→    const machinationsResults = JSON.stringify(machinationsResult);
   973→    const pathologies = JSON.stringify({
   974→      transitive_overpowered: transitiveResult.overpowered,
   975→      transitive_underpowered: transitiveResult.underpowered,
   976→      machinations_pathologies: machinationsResult.detected_pathologies,
   977→      stability_pathology_risks: stability.pathology_risks,
   978→    });
   979→    const corrections = JSON.stringify({
   980→      transitive: transitiveResult.suggestions,
   981→      intransitive: intransitiveResult.suggestions,
   982→      monte_carlo: monteCarloResult.suggestions,
   983→      machinations: machinationsResult.recommendations,
   984→    });
   985→    const situationalValues = JSON.stringify(situationalResult);
   986→    const fullResult = JSON.stringify(result);
   987→
   988→    const overallBalanceScore = stability.overall_stability;
   989→    const imbalanceCount =
   990→      transitiveResult.overpowered.length +
   991→      transitiveResult.underpowered.length +
   992→      machinationsResult.detected_pathologies.length;
   993→
   994→    await db.projectBalanceResult.upsert({
   995→      where: { projectId: proj.id },
   996→      create: {
   997→        projectId: proj.id,
   998→        balanceType: balanceType,
   999→        overallBalanceScore,
  1000→        imbalanceCount,
  1001→        elementCount: objects.length,
  1002→        inputData,
  1003→        elements: elementsData,
  1004→        costPowerCurves,
  1005→        intransitiveMatrix,
  1006→        nashEquilibrium,
  1007→        monteCarloResults,
  1008→        machinationsResults,
  1009→        pathologies,
  1010→        corrections,
  1011→        situationalValues,
  1012→        fullResult,
  1013→      },
  1014→      update: {
  1015→        balanceType: balanceType,
  1016→        overallBalanceScore,
  1017→        imbalanceCount,
  1018→        elementCount: objects.length,
  1019→        inputData,
  1020→        elements: elementsData,
  1021→        costPowerCurves,
  1022→        intransitiveMatrix,
  1023→        nashEquilibrium,
  1024→        monteCarloResults,
  1025→        machinationsResults,
  1026→        pathologies,
  1027→        corrections,
  1028→        situationalValues,
  1029→        fullResult,
  1030→      },
  1031→    });
  1032→
  1033→    await updateProjectStage(proj.id, "balance");
  1034→
  1035→    // safeJsonParse is imported but unused — satisfy linter
  1036→    void safeJsonParse;
  1037→
  1038→    return NextResponse.json(result);
  1039→  } catch (error) {
  1040→    console.error("[balance/analyze] error:", error);
  1041→    return SERVER_ERROR();
  1042→  }
  1043→}
