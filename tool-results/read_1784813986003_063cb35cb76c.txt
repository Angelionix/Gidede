    76→
    77→function ChatMessage({ msg }: { msg: ChatMsg }) {
    78→  const isUser = msg.role === "user";
    79→  const isSystem = msg.role === "system";
    80→
    81→  return (
    82→    <div
    83→      className={`flex gap-2 ${isUser ? "justify-end" : "justify-start"}`}
    84→    >
    85→      {!isUser && (
    86→        <div className="shrink-0 w-7 h-7 rounded-full bg-primary/10 flex items-center justify-center mt-1">
    87→          {isSystem ? (
    88→            <Info className="h-3.5 w-3.5 text-muted-foreground" />
    89→          ) : (
    90→            <Bot className="h-3.5 w-3.5 text-primary" />
    91→          )}
    92→        </div>
    93→      )}
    94→      <div
    95→        className={`max-w-[80%] rounded-lg px-3 py-2 text-sm ${
    96→          isUser
    97→            ? "bg-primary text-primary-foreground"
    98→            : isSystem
    99→            ? "bg-muted/50 text-muted-foreground italic"
   100→            : "bg-muted"
   101→        }`}
   102→      >
   103→        <div className="whitespace-pre-wrap break-words">
   104→          {msg.content}
   105→          {msg.isStreaming && (
   106→            <span className="inline-block w-1.5 h-4 bg-primary/60 animate-pulse ml-0.5 align-text-bottom" />
   107→          )}
   108→        </div>
   109→        {msg.metadata?.model_used && !isUser && !msg.isStreaming && (
   110→          <div className="mt-1 text-[10px] opacity-60">
   111→            {(msg.metadata.model_used as string) || ""} • {(msg.metadata.provider as string) || ""} • {((msg.metadata.latency_ms as number) || 0)}ms
   112→          </div>
   113→        )}
   114→      </div>
   115→      {isUser && (
   116→        <div className="shrink-0 w-7 h-7 rounded-full bg-primary flex items-center justify-center mt-1">
   117→          <MessageSquare className="h-3.5 w-3.5 text-primary-foreground" />
   118→        </div>
   119→      )}
   120→    </div>
   121→  );
   122→}

   307→function ChatHistoryList({
   308→  messages,
   309→  onLoadMore,
   310→  hasMore,
   311→}: {
   312→  messages: ChatMsg[];
   313→  onLoadMore: () => void;
   314→  hasMore: boolean;
   315→}) {
   316→  // Group messages by date
   317→  const grouped = messages.reduce<Record<string, ChatMsg[]>>((acc, msg) => {
   318→    const date = new Date(msg.timestamp).toLocaleDateString("ru-RU", {
   319→      day: "numeric",
   320→      month: "long",
   321→    });
   322→    if (!acc[date]) acc[date] = [];
   323→    acc[date].push(msg);
   324→    return acc;
   325→  }, {});
   326→
   327→  return (
   328→    <div className="space-y-4">
   329→      {Object.entries(grouped).map(([date, msgs]) => (
   330→        <div key={date}>
   331→          <div className="text-[10px] text-muted-foreground uppercase tracking-wider mb-2">
   332→            {date}
   333→          </div>
   334→          <div className="space-y-2">
   335→            {msgs.map((msg) => (
   336→              <div
   337→                key={msg.id}
   338→                className="flex items-start gap-2 p-2 rounded-md hover:bg-muted/50 cursor-pointer"
   339→              >
   340→                <div className="shrink-0 mt-0.5">
   341→                  {msg.role === "user" ? (
   342→                    <MessageSquare className="h-3 w-3 text-muted-foreground" />
   343→                  ) : (
   344→                    <Bot className="h-3 w-3 text-primary" />
   345→                  )}
   346→                </div>
   347→                <div className="flex-1 min-w-0">
   348→                  <p className="text-xs line-clamp-2">{msg.content}</p>
   349→                  <span className="text-[10px] text-muted-foreground">
   350→                    {new Date(msg.timestamp).toLocaleTimeString("ru-RU", {
   351→                      hour: "2-digit",
   352→                      minute: "2-digit",
   353→                    })}
   354→                  </span>
   355→                </div>
   356→              </div>
   357→            ))}
   358→          </div>
   359→        </div>
   360→      ))}
   361→      {hasMore && (
   362→        <Button variant="ghost" size="sm" onClick={onLoadMore} className="w-full text-xs">
   363→          Загрузить ещё
   364→        </Button>
   365→      )}
   366→    </div>
   367→  );
   368→}

   370→// ============================================================
   371→// Main Block 7 Page — AI Assistant with SSE Streaming
   372→// ============================================================
   373→
   374→export default function Block7Page() {
   375→  const { apiFetch, token } = useAuth();
   376→  const { toast } = useToast();
   377→
   378→  // --- Pipeline ---
   379→  const projectId =
   380→    typeof window !== "undefined"
   381→      ? localStorage.getItem("gidede_active_project")
   382→      : null;
   383→  const pipeline = usePipeline(projectId);
   384→
   385→  // --- Chat state ---
   386→  const [messages, setMessages] = useState<ChatMsg[]>([]);
   387→  const [inputValue, setInputValue] = useState("");
   388→  const [isSending, setIsSending] = useState(false);
   389→  const [isStreaming, setIsStreaming] = useState(false);
   390→  const chatEndRef = useRef<HTMLDivElement>(null);
   391→  const abortControllerRef = useRef<AbortController | null>(null);
   392→
   393→  // --- Suggestions state ---
   394→  const [suggestions, setSuggestions] = useState<Suggestion[]>([]);
   395→  const [suggestionsBlock, setSuggestionsBlock] = useState(1);
   396→  const [isSuggestionsLoading, setIsSuggestionsLoading] = useState(false);
   397→
   398→  // --- Alerts state ---
   399→  const [alerts, setAlerts] = useState<Alert[]>([]);
   400→  const [isAlertsLoading, setIsAlertsLoading] = useState(false);
   401→
   402→  // --- Chat history state ---
   403→  const [chatHistory, setChatHistory] = useState<ChatMsg[]>([]);
   404→  const [hasMoreHistory, setHasMoreHistory] = useState(false);
   405→  const [historyLimit, setHistoryLimit] = useState(50);
   406→
   407→  // --- Tab state ---
   408→  const [mainTab, setMainTab] = useState("chat");
   409→
   410→  // --- Auto-scroll ---
   411→  useEffect(() => {
   412→    chatEndRef.current?.scrollIntoView({ behavior: "smooth" });
   413→  }, [messages]);
   414→
   415→  // --- Load chat history on mount ---
   416→  useEffect(() => {
   417→    if (projectId && mainTab === "chat") {
   418→      loadChatHistory();
   419→    }
   420→    // eslint-disable-next-line react-hooks/exhaustive-deps
   421→  }, [projectId, mainTab]);
   422→
   423→  // --- Load chat history from server ---
   424→  const loadChatHistory = useCallback(async () => {
   425→    try {
   426→      const data = await apiFetch<{
   427→        messages: { id: string; role: string; content: string; timestamp: number; metadata?: Record<string, unknown> }[];
   428→        total: number;
   429→      }>(
   430→        `/assistant/history${projectId ? `?project_id=${projectId}` : ""}&limit=${historyLimit}`
   431→      );
   432→      const historyMsgs: ChatMsg[] = (data.messages || []).map((m) => ({
   433→        id: m.id,
   434→        role: m.role as "user" | "assistant" | "system",
   435→        content: m.content,
   436→        timestamp: m.timestamp,
   437→        metadata: m.metadata,
   438→      }));
   439→      setChatHistory(historyMsgs);
   440→      setHasMoreHistory(data.total > historyLimit);
   441→      // If local messages empty, populate from history
   442→      if (messages.length === 0 && historyMsgs.length > 0) {
   443→        setMessages(historyMsgs);
   444→      }
   445→    } catch {
   446→      // Silently ignore — history is optional
   447→    }
   448→  }, [projectId, historyLimit, apiFetch, messages.length]);
   449→
   450→  // --- SSE Streaming send ---
   451→  const handleSendStream = useCallback(async () => {
   452→    const msg = inputValue.trim();
   453→    if (!msg || isSending) return;
   454→
   455→    setIsSending(true);
   456→    setIsStreaming(true);
   457→    setInputValue("");
   458→
   459→    // Add user message
   460→    const userMsg: ChatMsg = {
   461→      id: Date.now().toString(),
   462→      role: "user",
   463→      content: msg,
   464→      timestamp: Date.now(),
   465→    };
   466→    setMessages((prev) => [...prev, userMsg]);
   467→
   468→    // Create streaming assistant message
   469→    const streamMsgId = (Date.now() + 1).toString();
   470→    const streamMsg: ChatMsg = {
   471→      id: streamMsgId,
   472→      role: "assistant",
   473→      content: "",
   474→      timestamp: Date.now(),
   475→      isStreaming: true,
   476→    };
   477→    setMessages((prev) => [...prev, streamMsg]);
   478→
   479→    try {
   480→      const baseUrl = getApiBaseUrl();
   481→      const abortController = new AbortController();
   482→      abortControllerRef.current = abortController;
   483→
   484→      const response = await fetch(`${baseUrl}/assistant/chat/stream`, {
   485→        method: "POST",
   486→        headers: {
   487→          "Content-Type": "application/json",
   488→          ...(token ? { Authorization: `Bearer ${token}` } : {}),
   489→        },
   490→        body: JSON.stringify({
   491→          message: msg,
   492→          project_id: projectId || undefined,
   493→          context: undefined,
   494→        }),
   495→        signal: abortController.signal,
   496→      });
   497→
   498→      if (!response.ok) {
   499→        throw new Error(`HTTP ${response.status}: ${response.statusText}`);
   500→      }
   501→
   502→      const reader = response.body?.getReader();
   503→      if (!reader) {
   504→        throw new Error("No response body for streaming");
   505→      }
   506→
   507→      const decoder = new TextDecoder();
   508→      let accumulatedContent = "";
   509→
   510→      while (true) {
   511→        const { done, value } = await reader.read();
   512→        if (done) break;
   513→
   514→        const chunk = decoder.decode(value, { stream: true });
   515→        // Parse SSE events
   516→        const lines = chunk.split("\n");
   517→        for (const line of lines) {
   518→          if (line.startsWith("data: ")) {
   519→            const jsonStr = line.slice(6).trim();
   520→            if (!jsonStr || jsonStr === "[DONE]") continue;
   521→
   522→            try {
   523→              const event = JSON.parse(jsonStr);
   524→
   525→              if (event.type === "message") {
   526→                accumulatedContent = event.content || accumulatedContent;
   527→                setMessages((prev) =>
   528→                  prev.map((m) =>
   529→                    m.id === streamMsgId
   530→                      ? { ...m, content: accumulatedContent, isStreaming: true }
   531→                      : m
   532→                  )
   533→                );
   534→              } else if (event.type === "done") {
   535→                // Finalize the message with metadata
   536→                setMessages((prev) =>
   537→                  prev.map((m) =>
   538→                    m.id === streamMsgId
   539→                      ? {
   540→                          ...m,
   541→                          content: accumulatedContent,
   542→                          isStreaming: false,
   543→                          metadata: {
   544→                            model_used: event.model_used || "",
   545→                            provider: event.provider || "",
   546→                            latency_ms: event.latency_ms || 0,
   547→                          },
   548→                        }
   549→                      : m
   550→                  )
   551→                );
   552→              }
   553→            } catch {
   554→              // Not JSON, accumulate as raw text
   555→              accumulatedContent += jsonStr;
   556→              setMessages((prev) =>
   557→                prev.map((m) =>
   558→                  m.id === streamMsgId
   559→                    ? { ...m, content: accumulatedContent, isStreaming: true }
   560→                    : m
   561→                )
   562→              );
   563→            }
   564→          }
   565→        }
   566→      }
   567→
   568→      // Ensure message is finalized
   569→      setMessages((prev) =>
   570→        prev.map((m) =>
   571→          m.id === streamMsgId ? { ...m, isStreaming: false } : m
   572→        )
   573→      );
   574→    } catch (err) {
   575→      // If abort, just mark as stopped
   576→      if (err instanceof DOMException && err.name === "AbortError") {
   577→        setMessages((prev) =>
   578→          prev.map((m) =>
   579→            m.id === streamMsgId ? { ...m, isStreaming: false } : m
   580→          )
   581→        );
   582→      } else {
   583→        // Fallback: try non-streaming endpoint
   584→        try {
   585→          const data = await apiFetch<{
   586→            reply: string;
   587→            model_used: string;
   588→            provider: string;
   589→            latency_ms: number;
   590→          }>("/assistant/chat", {
   591→            method: "POST",
   592→            headers: { "Content-Type": "application/json" },
   593→            body: JSON.stringify({
   594→              message: msg,
   595→              project_id: projectId || undefined,
   596→              context: undefined,
   597→            }),
   598→          });
   599→
   600→          setMessages((prev) =>
   601→            prev.map((m) =>
   602→              m.id === streamMsgId
   603→                ? {
   604→                    ...m,
   605→                    content: data.reply,
   606→                    isStreaming: false,
   607→                    metadata: {
   608→                      model_used: data.model_used,
   609→                      provider: data.provider,
   610→                      latency_ms: data.latency_ms,
   611→                    },
   612→                  }
   613→                : m
   614→            )
   615→          );
   616→        } catch (fallbackErr) {
   617→          setMessages((prev) =>
   618→            prev.map((m) =>
   619→              m.id === streamMsgId
   620→                ? {
   621→                    ...m,
   622→                    content: `Ошибка: ${fallbackErr instanceof Error ? fallbackErr.message : "Не удалось отправить сообщение"}`,
   623→                    isStreaming: false,
   624→                    role: "system",
   625→                  }
   626→                : m
   627→            )
   628→          );
   629→        }
   630→      }
   631→    } finally {
   632→      setIsSending(false);
   633→      setIsStreaming(false);
   634→      abortControllerRef.current = null;
   635→    }
   636→  }, [inputValue, isSending, projectId, apiFetch, token]);
   637→
   638→  // --- Stop streaming ---
   639→  const handleStopStream = useCallback(() => {
   640→    if (abortControllerRef.current) {
   641→      abortControllerRef.current.abort();
   642→    }
   643→  }, []);
   644→
   645→  // --- Load suggestions ---
   646→  const handleLoadSuggestions = useCallback(async () => {
   647→    setIsSuggestionsLoading(true);
   648→    try {
   649→      const data = await apiFetch<{
   650→        block_id: number;
   651→        suggestions: Suggestion[];
   652→      }>(
   653→        `/assistant/suggestions?block_id=${suggestionsBlock}${projectId ? `&project_id=${projectId}` : ""}`
   654→      );
   655→      setSuggestions(data.suggestions);
   656→    } catch (err) {
   657→      toast({
   658→        title: "Ошибка загрузки подсказок",
   659→        description: err instanceof Error ? err.message : "Неизвестная ошибка",
   660→        variant: "destructive",
   661→      });
   662→    } finally {
   663→      setIsSuggestionsLoading(false);
   664→    }
   665→  }, [suggestionsBlock, projectId, apiFetch, toast]);
   666→
   667→  // --- Load alerts ---
   668→  const handleLoadAlerts = useCallback(async () => {
   669→    setIsAlertsLoading(true);
   670→    try {
   671→      const data = await apiFetch<{ alerts: Alert[]; total: number }>(
   672→        `/assistant/alerts${projectId ? `?project_id=${projectId}` : ""}`
   673→      );
   674→      setAlerts(data.alerts);
   675→    } catch (err) {
   676→      toast({
   677→        title: "Ошибка проверки",
   678→        description: err instanceof Error ? err.message : "Неизвестная ошибка",
   679→        variant: "destructive",
   680→      });
   681→    } finally {
   682→      setIsAlertsLoading(false);
   683→    }
   684→  }, [projectId, apiFetch, toast]);
   685→
   686→  // --- Clear history ---
   687→  const handleClearHistory = useCallback(async () => {
   688→    try {
   689→      await apiFetch("/assistant/history/clear", {
   690→        method: "POST",
   691→        headers: { "Content-Type": "application/json" },
   692→        body: JSON.stringify({
   693→          project_id: projectId || undefined,
   694→        }),
   695→      });
   696→      setMessages([]);
   697→      setChatHistory([]);
   698→      toast({ title: "История очищена" });
   699→    } catch {
   700→      // Silently clear local state
   701→      setMessages([]);
   702→      setChatHistory([]);
   703→    }
   704→  }, [projectId, apiFetch, toast]);
   705→
   706→  // --- Handle Enter key ---
   707→  const handleKeyDown = useCallback(
   708→    (e: React.KeyboardEvent) => {
   709→      if (e.key === "Enter" && !e.shiftKey) {
   710→        e.preventDefault();
   711→        handleSendStream();
   712→      }
   713→    },
   714→    [handleSendStream]
   715→  );
   716→
   717→  // --- Load more history ---
   718→  const handleLoadMoreHistory = useCallback(() => {
   719→    setHistoryLimit((prev) => prev + 50);
   720→  }, []);
   721→
   722→  return (
   723→    <div className="p-4 md:p-6 max-w-6xl mx-auto space-y-4">
   724→      {/* Header */}
   725→      <div className="flex items-center gap-3">
   726→        <Bot className="h-6 w-6 text-primary" />
   727→        <div>
   728→          <h1 className="text-2xl font-bold">AI-ассистент</h1>
   729→          <p className="text-sm text-muted-foreground">
   730→            Блок 7 • Спецификация 3.9 • SSE Streaming
   731→          </p>
   732→        </div>
   733→        {pipeline.pipelineState && (
   734→          <Badge variant="outline" className="ml-auto text-xs">
   735→            <Clock className="h-3 w-3 mr-1" />
   736→            {pipeline.pipelineState.current_block
   737→              ? `Блок ${pipeline.pipelineState.current_block}`
   738→              : "Пайплайн готов"}
   739→          </Badge>
   740→        )}
   741→      </div>
   742→
   743→      {/* Main Tabs */}
   744→      <Tabs value={mainTab} onValueChange={setMainTab}>
   745→        <TabsList className="grid w-full grid-cols-2 sm:grid-cols-4">
   746→          <TabsTrigger value="chat" className="flex items-center gap-1.5">
   747→            <MessageSquare className="h-4 w-4" />
   748→            <span className="hidden sm:inline">Чат</span>
   749→          </TabsTrigger>
   750→          <TabsTrigger value="suggestions" className="flex items-center gap-1.5">
   751→            <Lightbulb className="h-4 w-4" />
   752→            <span className="hidden sm:inline">Подсказки</span>
   753→          </TabsTrigger>
   754→          <TabsTrigger value="alerts" className="flex items-center gap-1.5">
   755→            <AlertTriangle className="h-4 w-4" />
   756→            <span className="hidden sm:inline">Уведомления</span>
   757→          </TabsTrigger>
   758→          <TabsTrigger value="history" className="flex items-center gap-1.5">
   759→            <Clock className="h-4 w-4" />
   760→            <span className="hidden sm:inline">История</span>
   761→          </TabsTrigger>
   762→        </TabsList>
   763→
   764→        {/* ====================== CHAT TAB ====================== */}
   765→        <TabsContent value="chat" className="mt-4">
   766→          <Card className="flex flex-col animate-fade-in" style={{ height: "calc(100vh - 240px)", minHeight: "400px" }}>
   767→            {/* Messages */}
   768→            <CardContent className="flex-1 overflow-y-auto p-4 space-y-3">
   769→              {messages.length === 0 && (
   770→                <div className="text-center text-muted-foreground py-12">
   771→                  <Bot className="h-12 w-12 mx-auto mb-3 opacity-20" />
   772→                  <p className="text-sm font-medium">AI-ассистент Gidede</p>
   773→                  <p className="text-xs mt-1 max-w-sm mx-auto">
   774→                    Задайте вопрос о геймдизайне, получите рекомендации по проекту
   775→                    или попросите проверить дизайн на проблемы.
   776→                  </p>
   777→                  <div className="flex flex-wrap gap-2 justify-center mt-4">
   778→                    {[
   779→                      "Какие механики подходят для RPG?",
   780→                      "Проверь мой Core Loop на патологии",
   781→                      "Как сбалансировать экономику?",
   782→                      "Что такое лудонарративный диссонанс?",
   783→                    ].map((q) => (
   784→                      <Button
   785→                        key={q}
   786→                        variant="outline"
   787→                        size="sm"
   788→                        className="text-xs h-7"
   789→                        onClick={() => {
   790→                          setInputValue(q);
   791→                        }}
   792→                      >
   793→                        {q}
   794→                      </Button>
   795→                    ))}
   796→                  </div>
   797→                </div>
   798→              )}
   799→
   800→              {messages.map((msg) => (
   801→                <ChatMessage key={msg.id} msg={msg} />
   802→              ))}
   803→
   804→              {isSending && !isStreaming && (
   805→                <div className="flex gap-2 items-center text-sm text-muted-foreground" role="status" aria-busy="true">
   806→                  <Bot className="h-5 w-5 text-primary animate-pulse-subtle" />
   807→                  <span>AI-ассистент думает...</span>
   808→                </div>
   809→              )}
   810→
   811→              <div ref={chatEndRef} />
   812→            </CardContent>
   813→
   814→            {/* Input */}
   815→            <div className="border-t p-3 flex gap-2">
   816→              <Input
   817→                value={inputValue}
   818→                onChange={(e) => setInputValue(e.target.value)}
   819→                onKeyDown={handleKeyDown}
   820→                placeholder="Задайте вопрос о геймдизайне..."
   821→                disabled={isSending}
   822→                className="flex-1"
   823→              />
   824→              {isStreaming ? (
   825→                <Button
   826→                  onClick={handleStopStream}
   827→                  variant="destructive"
   828→                  size="icon"
   829→                  title="Остановить генерацию"
   830→                  aria-label="Остановить генерацию"
   831→                >
   832→                  <span className="text-xs font-bold">&#9632;</span>
   833→                </Button>
   834→              ) : (
   835→                <Button
   836→                  onClick={handleSendStream}
   837→                  disabled={isSending || !inputValue.trim()}
   838→                  size="icon"
   839→                  aria-label="Отправить сообщение"
   840→                >
   841→                  {isSending ? (
   842→                    <Loader2 className="h-4 w-4 animate-spin" />
   843→                  ) : (
   844→                    <Send className="h-4 w-4" />
   845→                  )}
   846→                </Button>
   847→              )}
   848→              <Button
   849→                variant="ghost"
   850→                size="icon"
   851→                onClick={handleClearHistory}
   852→                title="Очистить историю"
   853→                aria-label="Очистить историю чата"
   854→              >
   855→                <Trash2 className="h-4 w-4" />
   856→              </Button>
   857→            </div>
   858→          </Card>
   859→        </TabsContent>
   860→
   861→        {/* ====================== SUGGESTIONS TAB ====================== */}
   862→        <TabsContent value="suggestions" className="mt-4">
   863→          <Card>
   864→            <CardHeader className="pb-3">
   865→              <CardTitle className="text-base flex items-center gap-2">
   866→                <Lightbulb className="h-5 w-5 text-primary" />
   867→                Контекстные подсказки
   868→              </CardTitle>
   869→            </CardHeader>
   870→            <CardContent className="space-y-4">
   871→              {/* Block selector */}
   872→              <div className="flex items-center gap-3">
   873→                <span className="text-sm font-medium">Блок:</span>
   874→                <div className="flex gap-1">
   875→                  {[1, 2, 3, 4, 5, 6, 7, 8].map((blockId) => (
   876→                    <Button
   877→                      key={blockId}
   878→                      variant={suggestionsBlock === blockId ? "default" : "outline"}
   879→                      size="sm"
   880→                      className="h-7 w-7 p-0 text-xs"
   881→                      onClick={() => setSuggestionsBlock(blockId)}
   882→                    >
   883→                      {blockId}
   884→                    </Button>
   885→                  ))}
   886→                </div>
   887→                <Button
   888→                  onClick={handleLoadSuggestions}
   889→                  disabled={isSuggestionsLoading}
   890→                  size="sm"
   891→                  className="ml-auto"
   892→                >
   893→                  {isSuggestionsLoading ? (
   894→                    <Loader2 className="h-3 w-3 animate-spin mr-1" />
   895→                  ) : (
   896→                    <Sparkles className="h-3 w-3 mr-1" />
   897→                  )}
   898→                  Загрузить
   899→                </Button>
   900→              </div>
   901→
   902→              <SuggestionsPanel
   903→                suggestions={suggestions}
   904→                isLoading={isSuggestionsLoading}
   905→                onRefresh={handleLoadSuggestions}
   906→              />
   907→            </CardContent>
   908→          </Card>
   909→        </TabsContent>
   910→
   911→        {/* ====================== ALERTS TAB ====================== */}
   912→        <TabsContent value="alerts" className="mt-4">
   913→          <Card>
   914→            <CardHeader className="pb-3">
   915→              <CardTitle className="text-base flex items-center gap-2">
   916→                <AlertTriangle className="h-5 w-5 text-primary" />
   917→                Проактивные уведомления
   918→              </CardTitle>
   919→            </CardHeader>
   920→            <CardContent>
   921→              <AlertsPanel
   922→                alerts={alerts}
   923→                isLoading={isAlertsLoading}
   924→                onRefresh={handleLoadAlerts}
   925→              />
   926→            </CardContent>
   927→          </Card>
   928→        </TabsContent>
   929→
   930→        {/* ====================== HISTORY TAB ====================== */}
   931→        <TabsContent value="history" className="mt-4">
   932→          <Card>
   933→            <CardHeader className="pb-3">
   934→              <div className="flex items-center justify-between">
   935→                <CardTitle className="text-base flex items-center gap-2">
   936→                  <Clock className="h-5 w-5 text-primary" />
   937→                  История чата
   938→                </CardTitle>
   939→                <Button
   940→                  variant="ghost"
   941→                  size="sm"
   942→                  onClick={handleClearHistory}
   943→                  className="h-7 text-xs text-destructive"
   944→                >
   945→                  <Trash2 className="h-3 w-3 mr-1" />
   946→                  Очистить
   947→                </Button>
   948→              </div>
   949→            </CardHeader>
   950→            <CardContent>
   951→              {chatHistory.length === 0 ? (
   952→                <p className="text-xs text-muted-foreground text-center py-8">
   953→                  Нет сохранённых сообщений
   954→                </p>
   955→              ) : (
   956→                <ChatHistoryList
   957→                  messages={chatHistory}
   958→                  onLoadMore={handleLoadMoreHistory}
   959→                  hasMore={hasMoreHistory}
   960→                />
   961→              )}
   962→            </CardContent>
   963→          </Card>
   964→        </TabsContent>
   965→      </Tabs>
   966→    </div>
   967→  );
   968→}
