   271→function deriveSectionContent(
   272→  sectionName: string,
   273→  project: ProjectData,
   274→  language: string
   275→): { content: string; source: string; requires_review: boolean } {
   276→  const isRu = language === "ru";
   277→  const name = project.name || "Untitled Project";
   278→  const genre = project.genre || "game";
   279→  const description = project.description || "";
   280→  const concept = project.concept;
   281→  const coreLoop = project.coreLoop;
   282→  const mda = project.mdaProfile;
   283→  const balance = project.balanceResult;
   284→  const progression = project.progression;
   285→  const economy = project.economy;
   286→
   287→  const placeholder = isRu
   288→    ? `Раздел «${sectionName}» в разработке.`
   289→    : `Section "${sectionName}" is under construction.`;
   290→
   291→  // Source helpers
   292→  const onePager = concept?.onePagerData
   293→    ? safeJsonParse<Record<string, unknown>>(concept.onePagerData, {})
   294→    : {};
   295→  const usp = concept?.usp || (onePager.usp as string) || "";
   296→  const aestheticProfile = concept?.aestheticProfile
   297→    ? safeJsonParse<Record<string, unknown>>(concept.aestheticProfile, {})
   298→    : {};
   299→  const dynamicsProfile = concept?.dynamicsProfile
   300→    ? safeJsonParse<Record<string, unknown>>(concept.dynamicsProfile, {})
   301→    : {};
   302→  const mechanicSet = concept?.mechanicSet
   303→    ? safeJsonParse<Record<string, unknown>>(concept.mechanicSet, {})
   304→    : {};
   305→  const coreSteps = coreLoop?.stepsData
   306→    ? safeJsonParse<unknown[]>(coreLoop.stepsData, [])
   307→    : [];
   308→  const mdaMechanics = mda?.mechanicSet
   309→    ? safeJsonParse<Record<string, unknown>>(mda.mechanicSet, {})
   310→    : {};
   311→
   312→  const fillFromSource = (
   313→    val: unknown,
   314→    fallback: string,
   315→    sourceLabel: string,
   316→    requiresReview = false
   317→  ): { content: string; source: string; requires_review: boolean } => {
   318→    if (val === null || val === undefined || val === "") {
   319→      return { content: fallback, source: "ai_generate", requires_review: true };
   320→    }
   321→    if (typeof val === "string") {
   322→      return { content: val, source: sourceLabel, requires_review: requiresReview };
   323→    }
   324→    return {
   325→      content: JSON.stringify(val, null, 2),
   326→      source: sourceLabel,
   327→      requires_review: requiresReview,
   328→    };
   329→  };
   330→
   331→  switch (sectionName) {
   332→    case "title":
   333→      return {
   334→        content: `# ${name}\n\n**Жанр:** ${genre}${description ? `\n\n${description}` : ""}`,
   335→        source: "auto_fill",
   336→        requires_review: false,
   337→      };
   338→    case "logline":
   339→      return {
   340→        content:
   341→          usp ||
   342→          (isRu
   343→            ? `${name} — это ${genre}, который предлагает уникальный опыт в своём жанре.`
   344→            : `${name} is a ${genre} that offers a unique experience in its genre.`),
   345→        source: concept?.usp ? "auto_fill" : "ai_generate",
   346→        requires_review: !concept?.usp,
   347→      };
   348→    case "concept":
   349→      return fillFromSource(
   350→        onePager.concept ||
   351→          (isRu
   352→            ? `Концепция игры «${name}» — ${description || "многогранный опыт в жанре " + genre}.`
   353→            : `Game concept of "${name}" — ${description || "a multifaceted " + genre + " experience."}`),
   354→        isRu
   355→          ? `Концепция «${name}» формируется. Опишите центральную идею игры.`
   356→          : `Concept of "${name}" is being formed. Describe the central idea of the game.`,
   357→        "auto_fill"
   358→      );
   359→    case "usp":
   360→      return fillFromSource(
   361→        usp,
   362→        isRu
   363→          ? `Уникальное торговое предложение «${name}» TBD.`
   364→          : `Unique selling proposition of "${name}" TBD.`,
   365→        "auto_fill"
   366→      );
   367→    case "core_loop":
   368→    case "core_loop_summary": {
   369→      if (coreSteps.length > 0) {
   370→        const steps = Array.isArray(coreSteps)
   371→          ? coreSteps
   372→              .map((s, i) => `${i + 1}. ${typeof s === "object" ? JSON.stringify(s) : String(s)}`)
   373→              .join("\n")
   374→          : String(coreSteps);
   375→        return {
   376→          content: `## Core Loop\n\n${steps}`,
   377→          source: "auto_fill",
   378→          requires_review: false,
   379→        };
   380→      }
   381→      return {
   382→        content: isRu
   383→          ? `Основной цикл «${name}» — типичная для жанра ${genre} последовательность действий.`
   384→          : `Core loop of "${name}" is a typical ${genre} action sequence.`,
   385→        source: "ai_generate",
   386→        requires_review: true,
   387→      };
   388→    }
   389→    case "mechanics_overview":
   390→    case "mechanics":
   391→    case "core_mechanics": {
   392→      if (Object.keys(mechanicSet).length > 0) {
   393→        return {
   394→          content: `## Mechanics\n\n\`\`\`json\n${JSON.stringify(mechanicSet, null, 2)}\n\`\`\``,
   395→          source: "auto_fill",
   396→          requires_review: false,
   397→        };
   398→      }
   399→      return {
   400→        content: isRu
   401→          ? `Механики «${name}» строятся вокруг жанровых конвенций ${genre}.`
   402→          : `Mechanics of "${name}" build around ${genre} genre conventions.`,
   403→        source: "ai_generate",
   404→        requires_review: true,
   405→      };
   406→    }
   407→    case "aesthetics":
   408→    case "art_style": {
   409→      const primary = mda?.primaryAesthetic || concept?.primaryAesthetic;
   410→      const aestheticData =
   411→        Object.keys(aestheticProfile).length > 0 ? aestheticProfile : null;
   412→      if (primary || aestheticData) {
   413→        return {
   414→          content: `## Aesthetics\n\n**Primary aesthetic:** ${primary || "—"}\n\n${
   415→            aestheticData
   416→              ? "```json\n" + JSON.stringify(aestheticData, null, 2) + "\n```"
   417→              : ""
   418→          }`,
   419→          source: "auto_fill",
   420→          requires_review: false,
   421→        };
   422→      }
   423→      return {
   424→        content: isRu
   425→          ? `Эстетика «${name}» определяется жанром ${genre}.`
   426→          : `Aesthetics of "${name}" are defined by the ${genre} genre.`,
   427→        source: "ai_generate",
   428→        requires_review: true,
   429→      };
   430→    }
   431→    case "balance":
   432→    case "balance_overview": {
   433→      if (balance) {
   434→        return {
   435→          content: `## Balance\n\n- Type: ${balance.balanceType || "—"}\n- Overall score: ${balance.overallBalanceScore ?? "—"}\n- Elements: ${balance.elementCount ?? "—"}\n- Imbalances: ${balance.imbalanceCount ?? "—"}`,
   436→          source: "auto_fill",
   437→          requires_review: false,
   438→        };
   439→      }
   440→      return {
   441→        content: isRu
   442→          ? `Баланс «${name}» TBD.`
   443→          : `Balance of "${name}" TBD.`,
   444→        source: "ai_generate",
   445→        requires_review: true,
   446→      };
   447→    }
   448→    case "progression": {
   449→      if (progression) {
   450→        return {
   451→          content: `## Progression\n\n- Total levels: ${progression.totalLevels ?? "—"}\n- Tier count: ${progression.tierCount ?? "—"}\n- Curve type: ${progression.curveType ?? "—"}\n- Target duration (h): ${progression.targetDurationHours ?? "—"}`,
   452→          source: "auto_fill",
   453→          requires_review: false,
   454→        };
   455→      }
   456→      return {
   457→        content: isRu
   458→          ? `Прогрессия «${name}» в разработке.`
   459→          : `Progression of "${name}" under development.`,
   460→        source: "ai_generate",
   461→        requires_review: true,
   462→      };
   463→    }
   464→    case "economy": {
   465→      if (economy) {
   466→        return {
   467→          content: `## Economy\n\n- System type: ${economy.systemType ?? "—"}\n- Resource count: ${economy.resourceCount ?? "—"}\n- Has pathology: ${economy.hasPathology ? "yes" : "no"}`,
   468→          source: "auto_fill",
   469→          requires_review: false,
   470→        };
   471→      }
   472→      return {
   473→        content: isRu
   474→          ? `Экономика «${name}» TBD.`
   475→          : `Economy of "${name}" TBD.`,
   476→        source: "ai_generate",
   477→        requires_review: true,
   478→      };
   479→    }
   480→    case "narrative":
   481→    case "world_overview":
   482→    case "characters":
   483→    case "plot_arcs":
   484→    case "themes":
   485→    case "tone_voice":
   486→    case "story_mechanics":
   487→    case "branching_structure": {
   488→      const ludonarrative = mda?.ludonarrativeCheck
   489→        ? safeJsonParse<Record<string, unknown>>(mda.ludonarrativeCheck, {})
   490→        : {};
   491→      if (Object.keys(ludonarrative).length > 0) {
   492→        return {
   493→          content: `## ${sectionName}\n\n${JSON.stringify(ludonarrative, null, 2)}`,
   494→          source: "ai_enrich",
   495→          requires_review: true,
   496→        };
   497→      }
   498→      return {
   499→        content: isRu
   500→          ? `Нарративный раздел «${sectionName}» для «${name}» требует ручного заполнения. Жанр: ${genre}.`
   501→          : `Narrative section "${sectionName}" for "${name}" requires manual content. Genre: ${genre}.`,
   502→        source: "manual",
   503→        requires_review: true,
   504→      };
   505→    }
   506→    case "target_audience":
   507→    case "experience_goals": {
   508→      const dynamicsText =
   509→        Object.keys(dynamicsProfile).length > 0
   510→          ? JSON.stringify(dynamicsProfile, null, 2)
   511→          : null;
   512→      return {
   513→        content: isRu
   514→          ? `Целевая аудитория «${name}» — игроки жанра ${genre}.${dynamicsText ? "\n\nПрофиль динамики:\n```json\n" + dynamicsText + "\n```" : ""}`
   515→          : `Target audience of "${name}" — players of the ${genre} genre.${dynamicsText ? "\n\nDynamics profile:\n```json\n" + dynamicsText + "\n```" : ""}`,
   516→        source: dynamicsText ? "auto_fill" : "ai_generate",
   517→        requires_review: !dynamicsText,
   518→      };
   519→    }
   520→    case "monetization":
   521→    case "market_position":
   522→    case "competitive_landscape": {
   523→      if (economy?.monetizationModel) {
   524→        const monet = safeJsonParse<Record<string, unknown>>(
   525→          economy.monetizationModel,
   526→          {}
   527→        );
   528→        return {
   529→          content: `## Monetization\n\n\`\`\`json\n${JSON.stringify(monet, null, 2)}\n\`\`\``,
   530→          source: "auto_fill",
   531→          requires_review: false,
   532→        };
   533→      }
   534→      return {
   535→        content: isRu
   536→          ? `Монетизация «${name}» TBD.`
   537→          : `Monetization of "${name}" TBD.`,
   538→        source: "ai_generate",
   539→        requires_review: true,
   540→      };
   541→    }
   542→    case "platforms": {
   543→      const platforms = onePager.platforms as string[] | undefined;
   544→      return {
   545→        content: Array.isArray(platforms) && platforms.length > 0
   546→          ? `## Platforms\n\n${platforms.map((p) => `- ${p}`).join("\n")}`
   547→          : isRu
   548→            ? `Платформы «${name}» TBD.`
   549→            : `Platforms of "${name}" TBD.`,
   550→        source: Array.isArray(platforms) && platforms.length > 0 ? "auto_fill" : "ai_generate",
   551→        requires_review: !(Array.isArray(platforms) && platforms.length > 0),
   552→      };
   553→    }
   554→    case "ux":
   555→    case "ux_flow":
   556→    case "ui_mockups":
   557→    case "tech_notes":
   558→    case "tech_stack":
   559→    case "tech_bible":
   560→    case "art_bible":
   561→    case "sound":
   562→    case "localization":
   563→    case "testing_plan":
   564→    case "risks":
   565→    case "team_fit":
   566→    case "live_ops_plan":
   567→    case "overview":
   568→      return {
   569→        content: placeholder,
   570→        source: "manual",
   571→        requires_review: true,
   572→      };
   573→    default:
   574→      return {
   575→        content: placeholder,
   576→        source: "manual",
   577→        requires_review: true,
   578→      };
   579→  }
   580→}

   582→function buildConsistencyReport(
   583→  sections: Record<string, { content: string; source: string; requires_review: boolean }>,
   584→  sectionOrder: string[]
   585→): {
   586→  issues: Array<{
   587→    severity: "error" | "warning" | "info";
   588→    section_a: string;
   589→    section_b: string;
   590→    issue_type: string;
   591→    description: string;
   592→    suggestion: string;
   593→  }>;
   594→  error_count: number;
   595→  warning_count: number;
   596→  info_count: number;
   597→  is_valid: boolean;
   598→} {
   599→  const issues: Array<{
   600→    severity: "error" | "warning" | "info";
   601→    section_a: string;
   602→    section_b: string;
   603→    issue_type: string;
   604→    description: string;
   605→    suggestion: string;
   606→  }> = [];
   607→
   608→  // Check 1: any section requiring review → warning
   609→  for (const key of sectionOrder) {
   610→    const sec = sections[key];
   611→    if (!sec) continue;
   612→    if (sec.requires_review) {
   613→      issues.push({
   614→        severity: "warning",
   615→        section_a: key,
   616→        section_b: "review",
   617→        issue_type: "incomplete_section",
   618→        description: `Section "${key}" requires manual review (source: ${sec.source}).`,
   619→        suggestion: "Заполните секцию вручную или уточните источник данных.",
   620→      });
   621→    }
   622→    if (!sec.content || sec.content.trim().length < 20) {
   623→      issues.push({
   624→        severity: "info",
   625→        section_a: key,
   626→        section_b: "content_length",
   627→        issue_type: "short_content",
   628→        description: `Section "${key}" has very short content (${sec.content?.length || 0} chars).`,
   629→        suggestion: "Расширьте описание секции.",
   630→      });
   631→    }
   632→  }
   633→
   634→  // Check 2: section_pair mismatches
   635→  if (sections["core_loop"] && sections["mechanics"]) {
   636→    if (
   637→      sections["core_loop"].source === "ai_generate" &&
   638→      sections["mechanics"].source === "auto_fill"
   639→    ) {
   640→      issues.push({
   641→        severity: "info",
   642→        section_a: "core_loop",
   643→        section_b: "mechanics",
   644→        issue_type: "consistency_drift",
   645→        description: "Core loop is AI-generated while mechanics is auto-filled.",
   646→        suggestion: "Синхронизируйте core loop с механиками.",
   647→      });
   648→    }
   649→  }
   650→  if (sections["aesthetics"] && sections["narrative"]) {
   651→    if (
   652→      sections["aesthetics"].requires_review &&
   653→      sections["narrative"].requires_review
   654→    ) {
   655→      issues.push({
   656→        severity: "warning",
   657→        section_a: "aesthetics",
   658→        section_b: "narrative",
   659→        issue_type: "incomplete_pair",
   660→        description: "Обе секции (aesthetics + narrative) требуют ручного заполнения.",
   661→        suggestion: "Заполните хотя бы одну из секций.",
   662→      });
   663→    }
   664→  }
   665→
   666→  const errors = issues.filter((i) => i.severity === "error").length;
   667→  const warnings = issues.filter((i) => i.severity === "warning").length;
   668→  const infos = issues.filter((i) => i.severity === "info").length;
   669→
   670→  return {
   671→    issues,
   672→    error_count: errors,
   673→    warning_count: warnings,
   674→    info_count: infos,
   675→    is_valid: errors === 0,
   676→  };
   677→}

   679→export async function POST(request: NextRequest) {
   680→  const startedAt = Date.now();
   681→  const user = await getCurrentUser(request);
   682→  if (!user) return UNAUTH();
   683→
   684→  try {
   685→    const body = await request.json().catch(() => ({}));
   686→    const projectId = body?.project_id?.toString().trim() || undefined;
   687→    const targetFormat = body?.target_format?.toString().trim() || "full_gdd";
   688→    const detailLevel = body?.detail_level?.toString().trim() || "standard";
   689→    const audience = body?.target_audience_doc?.toString().trim() || "team_sync";
   690→    const projectStage = body?.project_stage?.toString().trim() || "preproduction";
   691→    const language = body?.language?.toString().trim() || "ru";
   692→    const customSections = Array.isArray(body?.custom_sections)
   693→      ? body.custom_sections.map(String)
   694→      : [];
   695→    const excludedSections = Array.isArray(body?.excluded_sections)
   696→      ? body.excluded_sections.map(String)
   697→      : [];
   698→
   699→    if (!VALID_FORMATS.includes(targetFormat)) {
   700→      return NextResponse.json(
   701→        { detail: `Неверный формат: ${targetFormat}` },
   702→        { status: 422 }
   703→      );
   704→    }
   705→    if (!VALID_DETAIL.includes(detailLevel)) {
   706→      return NextResponse.json(
   707→        { detail: `Неверный уровень детализации: ${detailLevel}` },
   708→        { status: 422 }
   709→      );
   710→    }
   711→    if (!VALID_AUDIENCE.includes(audience)) {
   712→      return NextResponse.json(
   713→        { detail: `Неверная аудитория: ${audience}` },
   714→        { status: 422 }
   715→      );
   716→    }
   717→    if (!VALID_STAGES.includes(projectStage)) {
   718→      return NextResponse.json(
   719→        { detail: `Неверная стадия проекта: ${projectStage}` },
   720→        { status: 422 }
   721→      );
   722→    }
   723→
   724→    const owned = await getOwnedProject(user, projectId);
   725→    if (owned instanceof NextResponse) return owned;
   726→    const proj = owned.project as ProjectData;
   727→
   728→    // --- Section catalogue ---
   729→    let sectionsList = FORMAT_SECTIONS[targetFormat] || FORMAT_SECTIONS.full_gdd;
   730→    if (customSections.length > 0) {
   731→      sectionsList = [...new Set([...sectionsList, ...customSections])];
   732→    }
   733→    if (excludedSections.length > 0) {
   734→      sectionsList = sectionsList.filter((s) => !excludedSections.includes(s));
   735→    }
   736→
   737→    // --- Section mappings / readiness ---
   738→    const activeMappings: Record<string, unknown> = {};
   739→    const sectionReadiness: Record<string, unknown> = {};
   740→    const autoFillable: string[] = [];
   741→    const manualSections: string[] = [];
   742→    const aiGeneratable: string[] = [];
   743→
   744→    for (const sectionName of sectionsList) {
   745→      const filled = deriveSectionContent(sectionName, proj, language);
   746→      activeMappings[sectionName] = {
   747→        source: filled.source,
   748→        auto_fill: filled.source === "auto_fill",
   749→        ai_enrich: filled.source === "ai_enrich",
   750→        ai_generate: filled.source === "ai_generate",
   751→        ai_suggest: false,
   752→        manual: filled.source === "manual",
   753→        diagram: false,
   754→        tables: false,
   755→        formulas: false,
   756→      };
   757→
   758→      let readiness: string;
   759→      if (filled.source === "auto_fill") {
   760→        readiness = "ready";
   761→        autoFillable.push(sectionName);
   762→      } else if (filled.source === "ai_generate" || filled.source === "ai_enrich") {
   763→        readiness = "ai_generatable";
   764→        aiGeneratable.push(sectionName);
   765→      } else {
   766→        readiness = "manual_required";
   767→        manualSections.push(sectionName);
   768→      }
   769→      sectionReadiness[sectionName] = {
   770→        status: readiness,
   771→        coverage: filled.source === "auto_fill" ? 1.0 : filled.source === "manual" ? 0.0 : 0.5,
   772→        auto_fillable: filled.source === "auto_fill",
   773→      };
   774→    }
   775→
   776→    const detailFactor = DETAIL_FACTOR[detailLevel] || 1.0;
   777→    const coverageScore = autoFillable.length / Math.max(1, sectionsList.length);
   778→
   779→    const dataMapping = {
   780→      active_mappings: activeMappings,
   781→      section_readiness: sectionReadiness,
   782→      auto_fillable_sections: autoFillable,
   783→      manual_sections: manualSections,
   784→      ai_generatable_sections: aiGeneratable,
   785→      coverage_score: Number(coverageScore.toFixed(3)),
   786→    };
   787→
   788→    // --- Auto-filled sections ---
   789→    const sectionsContent: Record<string, {
   790→      content: string;
   791→      source: string;
   792→      auto_filled: boolean;
   793→      diagram?: string;
   794→      tables?: Record<string, unknown>[];
   795→      formulas?: string[];
   796→      requires_review: boolean;
   797→    }> = {};
   798→    for (const sectionName of sectionsList) {
   799→      const filled = deriveSectionContent(sectionName, proj, language);
   800→      // Adjust content length based on detail factor
   801→      let content = filled.content;
   802→      if (detailFactor > 1.5 && filled.source === "ai_generate") {
   803→        content =
   804→          content +
   805→          (language === "ru"
   806→            ? "\n\n_Расширенное описание для детального уровня:_ смежные аспекты, влияющие на эту секцию, включают кросс-дисциплинарные зависимости и долгосрочные последствия для проекта."
   807→            : "\n\n_Extended description for detailed level:_ related aspects affecting this section include cross-disciplinary dependencies and long-term consequences for the project.");
   808→      } else if (detailFactor < 0.7 && filled.source === "ai_generate") {
   809→        content = content.split("\n")[0];
   810→      }
   811→      sectionsContent[sectionName] = {
   812→        content,
   813→        source: filled.source,
   814→        auto_filled: filled.source === "auto_fill",
   815→        requires_review: filled.requires_review,
   816→      };
   817→    }
   818→
   819→    const autoFilledSections = {
   820→      sections: sectionsContent,
   821→      count: Object.keys(sectionsContent).length,
   822→      total_coverage: Number(coverageScore.toFixed(3)),
   823→    };
   824→
   825→    // --- AI enriched (placeholder; treat AI-generated as enriched too) ---
   826→    const enrichedSections: Record<string, unknown> = {};
   827→    const generatedSections: Record<string, unknown> = {};
   828→    for (const [k, v] of Object.entries(sectionsContent)) {
   829→      if (v.source === "ai_enrich") enrichedSections[k] = v;
   830→      if (v.source === "ai_generate") generatedSections[k] = v;
   831→    }
   832→    const aiEnriched = {
   833→      enriched_sections: enrichedSections,
   834→      generated_sections: generatedSections,
   835→      enriched_count: Object.keys(enrichedSections).length,
   836→      generated_count: Object.keys(generatedSections).length,
   837→      failed_sections: [],
   838→      total_coverage: Number(coverageScore.toFixed(3)),
   839→    };
   840→
   841→    // --- Manual skeletons ---
   842→    const skeletons: Record<string, {
   843→      section_name: string;
   844→      priority: "critical" | "important" | "optional";
   845→      template: string;
   846→      hints: string[];
   847→      estimated_effort: string;
   848→    }> = {};
   849→    const criticalSections: string[] = [];
   850→    const importantSections: string[] = [];
   851→    const optionalSections: string[] = [];
   852→    for (const sectionName of manualSections) {
   853→      const priority: "critical" | "important" | "optional" =
   854→        sectionName === "narrative" || sectionName === "art_style"
   855→          ? "important"
   856→          : "optional";
   857→      skeletons[sectionName] = {
   858→        section_name: sectionName,
   859→        priority,
   860→        template: `# ${sectionName}\n\n[Опишите секцию здесь]`,
   861→        hints: [
   862→          "Сфокусируйтесь на цели секции",
   863→          "Свяжите с существующими данными проекта",
   864→        ],
   865→        estimated_effort: priority === "critical" ? "1-2 дня" : "2-4 часа",
   866→      };
   867→      if (priority === "critical") criticalSections.push(sectionName);
   868→      else if (priority === "important") importantSections.push(sectionName);
   869→      else optionalSections.push(sectionName);
   870→    }
   871→
   872→    const manualSkeletons = {
   873→      skeletons,
   874→      critical_sections: criticalSections,
   875→      important_sections: importantSections,
   876→      optional_sections: optionalSections,
   877→      total_manual_count: manualSections.length,
   878→    };
   879→
   880→    // --- Assembled document ---
   881→    const assembledSections: Record<string, {
   882→      section_name: string;
   883→      content: string;
   884→      source: string;
   885→      has_diagram: boolean;
   886→      has_tables: boolean;
   887→      has_formulas: boolean;
   888→      requires_review: boolean;
   889→    }> = {};
   890→    for (const [k, v] of Object.entries(sectionsContent)) {
   891→      const hasDiagram = /```mermaid|```graph/.test(v.content);
   892→      const hasTables = /\|.*\|.*\n\|[-: |]+\|/.test(v.content);
   893→      const hasFormulas = /=|∑|∫|≤|≥/.test(v.content) && v.content.length > 0;
   894→      assembledSections[k] = {
   895→        section_name: k.replace(/_/g, " ").replace(/\b\w/g, (c) => c.toUpperCase()),
   896→        content: v.content,
   897→        source: v.source,
   898→        has_diagram: hasDiagram,
   899→        has_tables: hasTables,
   900→        has_formulas: hasFormulas,
   901→        requires_review: v.requires_review,
   902→      };
   903→    }
   904→    const filledSections = sectionsList.filter(
   905→      (s) => sectionsContent[s]?.content && sectionsContent[s]!.content.length > 0
   906→    ).length;
   907→    const consistencyReport = buildConsistencyReport(
   908→      Object.fromEntries(
   909→        Object.entries(sectionsContent).map(([k, v]) => [
   910→          k,
   911→          { content: v.content, source: v.source, requires_review: v.requires_review },
   912→        ])
   913→      ),
   914→      sectionsList
   915→    );
   916→
   917→    const assembledDocument = {
   918→      sections: assembledSections,
   919→      section_order: sectionsList,
   920→      consistency_report: consistencyReport,
   921→      total_sections: sectionsList.length,
   922→      filled_sections: filledSections,
   923→      coverage_score: Number(coverageScore.toFixed(3)),
   924→    };
   925→
   926→    // --- Formatted document (Markdown) ---
   927→    const title = proj.name || "Untitled Project";
   928→    const toc = sectionsList
   929→      .map((s, i) => `${i + 1}. [${s}](#${s})`)
   930→      .join("\n");
   931→    const markdownParts: string[] = [];
   932→    markdownParts.push(`# ${title}\n`);
   933→    markdownParts.push(`> Format: ${targetFormat} • Detail: ${detailLevel} • Stage: ${projectStage}\n`);
   934→    markdownParts.push(`## Table of Contents\n\n${toc}\n`);
   935→    for (const s of sectionsList) {
   936→      const sec = assembledSections[s];
   937→      if (sec) {
   938→        markdownParts.push(`## ${sec.section_name}\n\n${sec.content}\n`);
   939→      }
   940→    }
   941→    const markdown = markdownParts.join("\n");
   942→    const wordCount = markdown.split(/\s+/).filter(Boolean).length;
   943→    const estimatedPages = FORMAT_PAGE_ESTIMATES[targetFormat] || 10;
   944→
   945→    const formattedDocument = {
   946→      markdown,
   947→      title,
   948→      table_of_contents: toc,
   949→      section_count: sectionsList.length,
   950→      word_count: wordCount,
   951→      estimated_pages: estimatedPages,
   952→    };
   953→
   954→    const stagesCompleted = [1, 2, 3, 4, 5, 6];
   955→    const latencyMs = Date.now() - startedAt;
   956→
   957→    // --- Format spec ---
   958→    const exportFormats = ["md", "html", "pdf", "docx"];
   959→    const formatSpec = {
   960→      format: targetFormat,
   961→      detail_level: detailLevel,
   962→      sections: sectionsList,
   963→      estimated_pages: estimatedPages,
   964→      audience,
   965→      export_formats: exportFormats,
   966→    };
   967→
   968→    const profile = {
   969→      format_spec: formatSpec,
   970→      data_mapping: dataMapping,
   971→      auto_filled_sections: autoFilledSections,
   972→      ai_enriched_sections: aiEnriched,
   973→      manual_skeletons: manualSkeletons,
   974→      assembled_document: assembledDocument,
   975→      formatted_document: formattedDocument,
   976→      stages_completed: stagesCompleted,
   977→      coverage_score: Number(coverageScore.toFixed(3)),
   978→      latency_ms: latencyMs,
   979→    };
   980→
   981→    // --- Persist ---
   982→    await db.projectGDD.upsert({
   983→      where: { projectId: proj.id },
   984→      create: {
   985→        projectId: proj.id,
   986→        format: targetFormat,
   987→        sectionCount: sectionsList.length,
   988→        completenessPercent: coverageScore,
   989→        inputData: JSON.stringify({
   990→          target_format: targetFormat,
   991→          detail_level: detailLevel,
   992→          target_audience_doc: audience,
   993→          project_stage: projectStage,
   994→          language,
   995→        }),
   996→        sections: JSON.stringify(assembledSections),
   997→        visualElements: JSON.stringify({}),
   998→        consistencyIssues: JSON.stringify(consistencyReport.issues),
   999→        completenessReport: JSON.stringify({
  1000→          total_sections: sectionsList.length,
  1001→          auto_filled: autoFillable.length,
  1002→          ai_generated: aiGeneratable.length,
  1003→          manual_filled: 0,
  1004→          manual_pending: manualSections.length,
  1005→          completeness_percent: Number((coverageScore * 100).toFixed(1)),
  1006→        }),
  1007→        fullProfile: JSON.stringify(profile),
  1008→      },
  1009→      update: {
  1010→        format: targetFormat,
  1011→        sectionCount: sectionsList.length,
  1012→        completenessPercent: coverageScore,
  1013→        inputData: JSON.stringify({
  1014→          target_format: targetFormat,
  1015→          detail_level: detailLevel,
  1016→          target_audience_doc: audience,
  1017→          project_stage: projectStage,
  1018→          language,
  1019→        }),
  1020→        sections: JSON.stringify(assembledSections),
  1021→        visualElements: JSON.stringify({}),
  1022→        consistencyIssues: JSON.stringify(consistencyReport.issues),
  1023→        completenessReport: JSON.stringify({
  1024→          total_sections: sectionsList.length,
  1025→          auto_filled: autoFillable.length,
  1026→          ai_generated: aiGeneratable.length,
  1027→          manual_filled: 0,
  1028→          manual_pending: manualSections.length,
  1029→          completeness_percent: Number((coverageScore * 100).toFixed(1)),
  1030→        }),
  1031→        fullProfile: JSON.stringify(profile),
  1032→      },
  1033→    });
  1034→
  1035→    await updateProjectStage(proj.id, "gdd");
  1036→
  1037→    return NextResponse.json(profile);
  1038→  } catch (error) {
  1039→    console.error("[gdd/generate] error:", error);
  1040→    return SERVER_ERROR();
  1041→  }
