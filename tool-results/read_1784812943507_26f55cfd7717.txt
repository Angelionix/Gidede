   111→function statusBadge(status: string) {
   112→  switch (status) {
   113→    case "synced":
   114→      return "bg-green-100 text-green-800 dark:bg-green-900/30 dark:text-green-300";
   115→    case "synced_with_warnings":
   116→      return "bg-yellow-100 text-yellow-800 dark:bg-yellow-900/30 dark:text-yellow-300";
   117→    case "failed":
   118→      return "bg-red-100 text-red-800 dark:bg-red-900/30 dark:text-red-300";
   119→    default:
   120→      return "bg-muted text-muted-foreground";
   121→  }
   122→}

   132→  }
